export type ClusterEnvironmentHealth = "critical" | "warning" | "normal" | "offline";
export type ClusterEnvironmentFreshness = "current" | "expired";
export type ClusterEnvironmentFilter = "all" | ClusterEnvironmentHealth | "expired";

export interface ClusterEnvironmentMetric {
  label: string;
  value: string;
  tone: ClusterEnvironmentHealth | "normal";
}

export interface ClusterEnvironmentIncident {
  title: string;
  description: string;
  impact: string;
  duration: string;
  related: string;
}

export interface ClusterEnvironmentItem {
  id: string;
  name: string;
  location: string;
  health: ClusterEnvironmentHealth;
  freshness?: ClusterEnvironmentFreshness;
  metrics: readonly ClusterEnvironmentMetric[];
  incident?: ClusterEnvironmentIncident;
  reportLabel?: string;
}

export interface ClusterEnvironmentListProps extends Omit<ComponentPropsWithoutRef<"div">, "children"> {
  workspace?: OperationsWorkspaceOptions;
  environments?: readonly ClusterEnvironmentItem[];
  state?: "ready" | "loading" | "stale" | "error";
  refreshing?: boolean;
  retainDataOnError?: boolean;
  statusMessage?: string;
  onRefresh?: () => void;
  onRetry?: () => void | Promise<void>;
  onViewDetails?: (environment: ClusterEnvironmentItem) => void;
}

const healthPresentation: Record<ClusterEnvironmentHealth, { cardSurface: string; badge: BadgeStatus; label: string }> = {
  critical: { cardSurface: "bg-danger-surface-subtle", badge: "danger", label: "P0·严重" },

  offline: { cardSurface: "bg-neutral-status-surface", badge: "neutral", label: "离线" },

const healthIcons: Record<ClusterEnvironmentHealth, IconComponent> = {
  critical: createHugeIcon(Fire02Icon),
  warning: createHugeIcon(AlertSquareIcon),

const filterIcons: Partial<Record<ClusterEnvironmentFilter, IconComponent>> = healthIcons;

function EnvironmentCard({ item, onViewDetails }: { item: ClusterEnvironmentItem; onViewDetails?: (item: ClusterEnvironmentItem) => void }) {
  const presentation = healthPresentation[item.health];
  const Environment = useIcon("square-library");
  const strongFooterBadge = item.health !== "normal";
  return <Card data-health={item.health} data-freshness={item.freshness ?? "current"} className={cn("overflow-hidden rounded-2xl border-hairline border-border p-2", presentation.cardSurface)} label={`查看 ${item.name}`}>
    <CardContent className="flex min-h-0 flex-col gap-2 p-0">
      <div className="flex min-w-0 items-center gap-2 p-1">
        <span className="flex size-11 shrink-0 items-center justify-center rounded-lg bg-surface-floating text-fg-brand"><Environment size={22} strokeWidth={1.5} /></span>
        <div className="min-w-0"><p className="truncate text-label text-fg-subtle">{item.location}</p><h2 className="truncate text-body font-semibold text-fg-default">{item.name}</h2></div>
      </div>
      <dl className="grid grid-cols-3 rounded-xl bg-surface-floating p-1.5">
        {item.metrics.map((metric) => {
          const MetricIcon = healthIcons[metric.tone];

          return <div key={metric.label} className="min-w-0 px-1.5 py-1"><dt className="text-body text-fg-muted">{metric.label}</dt><dd className="mt-1 min-w-0 font-medium"><Badge variant="plain" status={healthPresentation[metric.tone].badge} aria-label={`${metric.label}：${healthPresentation[metric.tone].label}，${metric.value}`} leadingIcon={<MetricIcon strokeWidth={2} />}>{metric.value}</Badge></dd></div>;

    </CardContent>
    {item.freshness === "expired" && <div className="mt-2 px-1"><InlineNotice variant="emphasized" tone="warning"><InlineNoticeContent>{"数据已过期 · 以下为上次巡检结果"}</InlineNoticeContent></InlineNotice></div>}
    <CardFooter className="mt-2 flex-wrap justify-between gap-2 p-0"><InlineNotice variant="emphasized" tone={presentation.badge} className="min-w-0">{strongFooterBadge ? <Badge variant="strong" status={presentation.badge} size="sm" className="shrink-0">{presentation.label}</Badge> : <Badge status={presentation.badge} size="sm" className="shrink-0 border-0">{presentation.label}</Badge>}<InlineNoticeContent className="break-words">{item.incident?.title ?? item.reportLabel ?? "巡检 21 / 21 完成"}</InlineNoticeContent></InlineNotice><Button type="button" size="sm" variant="ghost" onClick={() => onViewDetails?.(item)}>查看详情</Button></CardFooter>
  </Card>;

/** A responsive, filterable cluster-environment overview composed entirely from Zeron primitives. */
export function ClusterEnvironmentList({ workspace, environments = defaultClusterEnvironments, className, state = "ready", refreshing = false, retainDataOnError = false, statusMessage, onRefresh, onRetry, onViewDetails, ...props }: ClusterEnvironmentListProps) {
  const [filter, setFilter] = useState<ClusterEnvironmentFilter>("all");
  const [query, setQuery] = useState("");
  const Search = useIcon("search");
  const List = useIcon("list");
  const Refresh = useIcon("rotate-ccw");
  const [pending, setPending] = useState<"refresh" | "retry" | null>(null);
  const inFlight = useRef(false);
  const [actionError, setActionError] = useState<{ data: typeof environments; message: string } | null>(null);
  const currentError = actionError?.data === environments ? actionError.message : undefined;
  const loading = state === "loading";
  const busy = loading || refreshing || pending !== null;
  const failed = state === "error" || currentError !== undefined;
  const replacesData = state === "error" && !retainDataOnError;
  async function runAction(kind: "refresh" | "retry") {
    const callback = kind === "refresh" ? onRefresh : onRetry;
    if (!callback || inFlight.current || loading || refreshing) return;
    inFlight.current = true;
    setPending(kind);
    setActionError(null);
    try { await callback(); }
    catch (cause) { setActionError({ data: environments, message: cause instanceof Error && cause.message ? cause.message : "集群环境加载失败" }); }
    finally { inFlight.current = false; setPending(null); }
  }
  const retryAction = onRetry ? <Button disabled={busy} loading={pending === "retry"} variant="secondary" onClick={() => void runAction("retry")}>重试</Button> : undefined;
  const feedback = loading ? <div role="status" aria-label="正在加载集群环境" className="space-y-3"><Skeleton className="h-8 w-40" /><Skeleton className="h-48 w-full" /></div>
    : replacesData ? <div className="flex min-w-0 w-full items-center justify-center min-h-[min(60vh,36rem)] px-6 py-12 sm:px-10 sm:py-16"><Alert status="danger" role={Boolean(currentError) ? "alert" : "group"} className="w-full max-w-xl"><AlertTitle>{"集群环境加载失败"}</AlertTitle><AlertDescription>{currentError ?? statusMessage}</AlertDescription><AlertAction>{retryAction}</AlertAction></Alert></div>
    : <>
      {failed ? <InlineNotice tone="danger" variant="emphasized" role={currentError ? "alert" : undefined}><InlineNoticeContent>集群环境刷新失败：{currentError ?? statusMessage ?? "请稍后重试。"} 当前显示上次获取的结果。</InlineNoticeContent>{retryAction && <InlineNoticeAction>{retryAction}</InlineNoticeAction>}</InlineNotice>
        : state === "stale" && <InlineNotice tone="warning" variant="emphasized"><InlineNoticeContent>{statusMessage ?? "环境数据已过期，当前显示上次获取的结果。"}</InlineNoticeContent></InlineNotice>}
      {busy && <InlineNotice tone="info" variant="emphasized"><span aria-hidden="true" className="inline-flex h-5 shrink-0 items-center [&_svg]:size-4"><span className="inline-flex animate-spin motion-reduce:animate-none"><Refresh /></span></span><InlineNoticeContent>{"正在刷新集群环境，保留上次获取的结果。"}</InlineNoticeContent></InlineNotice>}
    </>;
  const filters: readonly ClusterEnvironmentFilter[] = ["all", "critical", "warning", "normal", "offline", "expired"];
  const filterLabel: Record<ClusterEnvironmentFilter, string> = { all: "全部", critical: "严重", warning: "告警", normal: "正常", offline: "离线", expired: "数据过期" };
  const filtered = useMemo(() => environments.filter((item) => (filter === "all" || filter === "expired" ? filter === "all" || item.freshness === "expired" : item.health === filter) && `${item.name} ${item.location}`.toLocaleLowerCase().includes(query.trim().toLocaleLowerCase())), [environments, filter, query]);

  return <OperationsWorkspaceShell workspace={workspace} activeNavigation="clusters" title="集群环境" icon={List} className={cn("min-h-[42rem]", className)} {...props} aria-busy={busy || undefined}><PageContent><div className="shrink-0 border-b border-border-subtle p-3 pb-0"><div className="flex flex-wrap items-center justify-between gap-3 pb-3"><Tabs value={filter} onValueChange={(value) => setFilter(value as ClusterEnvironmentFilter)} variant="segment"><TabsList>{filters.map((value) => <TabItem key={value} value={value} label={filterLabel[value]} icon={filterIcons[value]} badge={value === "all" ? undefined : value === "expired" ? environments.filter((item) => item.freshness === "expired").length : environments.filter((item) => item.health === value).length} />)}</TabsList></Tabs><div className="flex w-full items-center gap-3 lg:w-auto"><label className="relative min-w-0 flex-1 lg:w-[28rem]"><span className="sr-only">搜索环境</span><Search className="pointer-events-none absolute left-2 top-1/2 size-4 -translate-y-1/2 text-fg-subtle" /><Input value={query} onChange={(event) => setQuery(event.target.value)} placeholder="搜索" className="pl-8" /></label><Button type="button" size="sm" variant="neutral" disabled={busy} loading={refreshing || pending === "refresh"} onClick={() => void runAction("refresh")}>刷新</Button></div></div></div><PageBody className="max-w-[101.25rem] p-3">{feedback}{!loading && !replacesData && <><div className="flex items-center justify-between gap-3 py-4 text-body text-fg-subtle"><span>当前共 {filtered.length} 个环境</span><span>{state === "stale" || failed ? "上次获取的结果" : "数据更新于 3 分钟前"}</span></div><CardGroup columns={3} separated proximityHover={false} className="items-start" style={{ gridTemplateColumns: "repeat(auto-fill, minmax(min(100%, 25.625rem), 1fr))" }}>{filtered.map((item) => <EnvironmentCard key={item.id} item={item} onViewDetails={onViewDetails} />)}</CardGroup>{filtered.length === 0 && <Empty reason={environments.length ? "no-filter-results" : "no-data"} scope="section"><EmptyHeader><EmptyTitle>{environments.length ? "没有符合条件的环境" : "暂无集群环境"}</EmptyTitle><EmptyDescription>{environments.length ? "调整搜索条件或清空筛选后重试。" : "添加集群环境后即可查看健康状态与巡检结果。"}</EmptyDescription></EmptyHeader>{environments.length > 0 && <EmptyActions><Button variant="secondary" onClick={() => { setQuery(""); setFilter("all"); }}>清空筛选</Button></EmptyActions>}</Empty>}</>}</PageBody></PageContent></OperationsWorkspaceShell>;
}
