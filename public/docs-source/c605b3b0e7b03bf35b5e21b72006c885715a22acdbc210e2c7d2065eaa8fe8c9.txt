export function ChatThread({
  children,
  composer,
  empty,
  busy = false,
  hasOlder = false,
  loadingOlder = false,
  onLoadOlder,
  bookmarkedIds,
  onBookmarkChange,
  bookmarkLabels,
  onRestore,
  composerRef,
  labels: overrides,
  locale = "en-US",
  className,
  ...props
}: ChatThreadProps) {
  const labels = useMemo(
    () => ({ ...chatThreadEnLabels, ...overrides }),
    [overrides],
  );
  const [entries, setEntries] = useState<ThreadEntry[]>([]);
  const [loadPending, setLoadPending] = useState(false);
  const [loadError, setLoadError] = useState(false);
  const [restorePreview, previewRestore] = useState<string | null>(null);
  const [pending, setPending] = useState(false);
  const mutationLock = useRef(false);
  const loadLock = useRef(false);
  const olderBusy = loadingOlder || loadPending;
  const olderBusyRef = useRef(olderBusy);
  olderBusyRef.current = olderBusy;
  const scroll = useThreadScroll(entries, children, olderBusy);
  const isEmpty = entries.length === 0;

  const register = useCallback((entry: ThreadEntry) => {
    setEntries((current) =>
      [...current.filter((item) => item.id !== entry.id), entry].sort((a, b) =>
        a.element.compareDocumentPosition(b.element) &
        Node.DOCUMENT_POSITION_FOLLOWING
          ? -1
          : 1,
      ),
    );
    return () =>
      setEntries((current) =>
        current.filter((item) => item.element !== entry.element),
      );
  }, []);

  const restore = useCallback(
    async (id: string, code: boolean) => {
      if (!onRestore || busy || olderBusy || loadLock.current || mutationLock.current) return false;
      mutationLock.current = true;
      setPending(true);
      try {
        const result = await onRestore(id, { code });
        previewRestore(null);
        // The checkpoint and its focus manager can unmount when the host commits.
        // Restore editor focus after that cleanup, using the public editor ref.
        requestAnimationFrame(() => {
          const textarea = composerRef?.current;
          textarea?.focus({ preventScroll: true });
          textarea?.setSelectionRange(
            textarea.value.length,
            textarea.value.length,
          );
        });
        const undo = result?.undo;
        toast.success(
          labels.restored,
          undo
            ? {
                action: {
                  label: labels.undo,
                  dismissOnClick: false,
                  onClick: async (notification) => {
                    if (mutationLock.current || loadLock.current || olderBusyRef.current) return;
                    mutationLock.current = true;
                    setPending(true);
                    try {
                      await undo();
                      toast.dismiss(notification.id);
                      toast.success(labels.undone);
                    } catch {
                      toast.error(labels.undoFailed);
                    } finally {
                      mutationLock.current = false;
                      setPending(false);
                    }
                  },
                },
              }
            : undefined,
        );
        return true;
      } catch {
        toast.error(labels.restoreFailed);
        return false;
      } finally {
        mutationLock.current = false;
        setPending(false);
      }
    },
    [onRestore, busy, olderBusy, composerRef, labels],
  );

  const loadOlder = async () => {
    if (!onLoadOlder || olderBusy || loadLock.current || mutationLock.current) return;
    loadLock.current = true;
    scroll.captureAnchor();
    setLoadPending(true);
    setLoadError(false);
    try {
      await onLoadOlder();
    } catch {
      setLoadError(true);
    } finally {
      loadLock.current = false;
      setLoadPending(false);
    }
  };

  const context = useMemo(
    () => ({
      labels,
      locale,
      busy: busy || olderBusy,
      pending,
      entries,
      restorePreview,
      previewRestore,
      register,
      pin: scroll.pin,
      restore: onRestore ? restore : undefined,
    }),
    [
      labels,
      locale,
      busy,
      olderBusy,
      pending,
      entries,
      restorePreview,
      register,
      scroll.pin,
      onRestore,
      restore,
    ],
  );
  const navigationItems = useMemo(() => {
    const bookmarks = new Set(bookmarkedIds);
    return entries.flatMap((entry, index) => {
      if (entry.role !== "user") return [];
      const next = entries[index + 1];
      return [{
        id: entry.id,
        label: entry.text || labels.conversation,
        bookmarked: bookmarks.has(entry.id),
        description: next?.role === "assistant" && next.text
          ? <Markdown text={next.text.slice(0, 640).split(/\n\s*\n/, 1)[0]} codeVariant="plain" /> : undefined,
      }];
    });
  }, [entries, bookmarkedIds, labels.conversation]);

  return (
    <ThreadContext.Provider value={context}>
      <SurfaceProvider role="floating">
      <div
        className={cn(
          "@container/thread relative flex h-full min-h-0 w-full flex-col bg-surface-floating text-fg-default",
          className,
        )}
        data-slot="chat-thread"
        {...props}
      >
        <ScrollArea
          className={cn("min-h-0 flex-1", isEmpty && "hidden")}
          contentProps={{ style: { minWidth: 0 } }}
          onWheelCapture={scroll.onUserIntent}
          onTouchStartCapture={scroll.onUserIntent}
          onPointerDownCapture={scroll.onUserIntent}
          onKeyDownCapture={(event) => {
            if (
              [
                "ArrowUp",
                "ArrowDown",
                "PageUp",
                "PageDown",
                "Home",
                "End",
                " ",
              ].includes(event.key)
            )
              scroll.onUserIntent();
          }}
          viewportRef={scroll.bindViewport}
          viewportClassName="overscroll-contain"
          viewportProps={{
            role: "region",
            "aria-label": labels.conversation,
            tabIndex: 0,
            onScroll: scroll.onScroll,
            style: { overflowAnchor: "none" },
          }}
        >
          <div
            ref={scroll.bindContent}
            className="relative mx-auto flex w-full max-w-3xl flex-col gap-6 px-4 pb-6 pt-4 @sm/thread:px-8"
          >
            {hasOlder && onLoadOlder && (
              <div className="flex flex-col items-center gap-2">
                <Button
                  variant="ghost"
                  size="sm"
                  onClick={loadOlder}
                  disabled={olderBusy || pending}
                  loading={olderBusy}
                >
                  {olderBusy ? labels.loadingOlder : labels.loadOlder}
                </Button>
                {loadError && (
                  <InlineNotice role="alert" variant="emphasized" tone="danger">
                    <InlineNoticeContent>{labels.loadFailed}</InlineNoticeContent>
                  </InlineNotice>
                )}
              </div>
            )}
            <div
              role="log"
              aria-label={labels.conversation}
              aria-live="polite"
              aria-relevant="additions"
              aria-busy={busy || olderBusy || pending}
              className="flex min-w-0 flex-col gap-6"
            >
              {children}
            </div>
            <div
              ref={scroll.spacerRef}
              aria-hidden="true"
              className="shrink-0"
            />
          </div>
        </ScrollArea>
        <div
          className={cn(
            "items-end justify-center px-6 pb-7 text-center",
            isEmpty ? "flex flex-1" : "hidden",
          )}
        >
          {empty ?? (
            <Empty reason="first-use" scope="inline" density="compact">
              <EmptyHeader><EmptyTitle as="h2">{labels.empty}</EmptyTitle></EmptyHeader>
            </Empty>
          )}
        </div>
        <div
          className="relative mx-auto w-full max-w-3xl shrink-0 px-4 pb-4 pt-2 @sm/thread:px-8"
          data-slot="chat-thread-composer"
        >
          {scroll.away && !isEmpty && (
            <ChatScrollToLatest className="bottom-full pb-4" label={labels.latest} onClick={scroll.latest} />
          )}
          {composer}
        </div>
        <div aria-hidden="true" className={isEmpty ? "flex-1" : "hidden"} />
        {navigationItems.length > 0 && (
          <MessageNavigation
            label={labels.navigation}
            activeId={scroll.activeId}
            onSelect={scroll.jump}
            onBookmarkChange={onBookmarkChange}
            bookmarkLabels={bookmarkLabels}
            getItemLabel={(item, index) => labels.message(index + 1, item.label)}
            items={navigationItems}
            className="absolute right-1 top-1/2 z-raised hidden max-h-64 -translate-y-1/2 @3xl/thread:flex"
          />
        )}
      </div>
      </SurfaceProvider>
    </ThreadContext.Provider>
  );
}

function useMessageEntry(
  id: string,
  role: "user" | "assistant",
  text = "",
  checkpoint?: ChatCheckpointData,
) {
  const context = useThread();
  const ref = useRef<HTMLElement>(null);
  const { register } = context;
  const checkpointRef = useRef(checkpoint);
  checkpointRef.current = checkpoint;
  const checkpointKey = JSON.stringify(checkpoint);
  useLayoutEffect(() => {
    if (ref.current)
      return register({
        id,
        role,
        text,
        element: ref.current,
        checkpoint: checkpointRef.current,
      });
  }, [register, id, role, text, checkpointKey]);
  const start = context.entries.findIndex(
    (entry) => entry.id === context.restorePreview,
  );
  const index = context.entries.findIndex((entry) => entry.id === id);
  return { ref, dimmed: start >= 0 && index >= start, context };
}

export function ChatCheckpoint({
  messageId,
  checkpoint,
}: {
  messageId: string;
  checkpoint: ChatCheckpointData;
}) {
  const context = useThread();
  const { labels } = context;
  const [open, setOpen] = useState(false);
  const triggerRef = useRef<HTMLButtonElement>(null);
  const restoredRef = useRef(false);
  const descriptionId = useId();
  const Pin = useIcon("pin");
  const index = context.entries.findIndex((entry) => entry.id === messageId);
  const ownFiles = [...new Set(checkpoint.files ?? [])];
  const files = [
    ...new Set(
      index < 0
        ? ownFiles
        : context.entries
            .slice(index)
            .flatMap((entry) => entry.checkpoint?.files ?? []),
    ),
  ];
  const affected = index < 0 ? 0 : context.entries.length - index;
  const { previewRestore } = context;
  useEffect(() => {
    if (!open) return;
    previewRestore(messageId);
    return () => previewRestore(null);
  }, [open, messageId, previewRestore]);
  const canRestore = !!context.restore;
  return (
    <div className="flex min-w-0 items-center gap-3 text-label text-fg-muted">
      <Separator
        tone="subtle"
        role="none"
        className="min-w-3 w-auto flex-1"
        aria-hidden="true"
      />
      <Pin size={12} />
      <span id={descriptionId} className="min-w-0 truncate">
        {checkpoint.label ?? labels.checkpoint}
        {ownFiles.length > 0 && ` · ${labels.changedFiles(ownFiles.length)}`}
      </span>
      {canRestore && (
        <Popover
          open={open}
          onOpenChange={(next) => {
            if (!context.pending) {
              restoredRef.current = false;
              setOpen(next);
            }
          }}
        >
          <PopoverTrigger
            ref={triggerRef}
            render={
              <Button
                variant="ghost"
                size="xs"
                disabled={context.busy || context.pending}
                aria-label={`${labels.restore}: ${context.entries[index]?.text ?? checkpoint.label ?? labels.checkpoint}`}
                aria-describedby={descriptionId}
              >
                {labels.restore}
              </Button>
            }
          />
          <PopoverContent
            align="end"
            className="max-w-xs"
            finalFocus={() =>
              restoredRef.current ? false : triggerRef.current
            }
          >
            <PopoverHeader>
              <PopoverTitle>{labels.restoreTitle}</PopoverTitle>
              <PopoverDescription>{labels.restoreImpact(affected, files.length)}</PopoverDescription>
            </PopoverHeader>
            {files.length > 0 && (
              <ul className="my-3 max-h-32 overflow-auto text-label text-fg-muted">
                {files.map((file) => (
                  <li key={file} className="break-all font-mono">
                    {file}
                  </li>
                ))}
              </ul>
            )}
            <PopoverFooter className="flex-col items-stretch">
              {files.length > 0 && (
                <Button
                  size="sm"
                  variant="neutral"
                  loading={context.pending}
                  disabled={context.pending}
                  onClick={async () => {
                    restoredRef.current = true;
                    if (await context.restore?.(checkpoint.id, true)) {
                      setOpen(false);
                    } else {
                      restoredRef.current = false;
                    }
                  }}
                >
                  {context.pending ? labels.restoring : labels.restoreCode}
                </Button>
              )}
              <Button
                size="sm"
                variant={files.length ? "secondary" : "neutral"}
                disabled={context.pending}
                onClick={async () => {
                  restoredRef.current = true;
                  if (await context.restore?.(checkpoint.id, false)) {
                    setOpen(false);
                  } else {
                    restoredRef.current = false;
                  }
                }}
              >
                {labels.restoreConversation}
              </Button>
              <Button
                size="sm"
                variant="ghost"
                disabled={context.pending}
                onClick={() => setOpen(false)}
              >
                {labels.cancel}
              </Button>
            </PopoverFooter>
          </PopoverContent>
        </Popover>
      )}
      <Separator
        tone="subtle"
        role="none"
        className="min-w-3 w-auto flex-1"
        aria-hidden="true"
      />
    </div>
  );
}

export function ChatUserMessage({

  branch,

/** The selected user message is included in removed so it can become the draft. */
export function splitThreadHistory<T extends { id: string }>(
  messages: readonly T[],
  messageId: string,
): { kept: T[]; removed: T[] } {
  const index = messages.findIndex((message) => message.id === messageId);
  return index < 0
    ? { kept: [...messages], removed: [] }
    : { kept: messages.slice(0, index), removed: messages.slice(index) };
}


"use client";

import { useCallback, useLayoutEffect, useRef, useState } from "react";
import { CHAT_SCROLL_FOLLOW_THRESHOLD } from "@zeron/ui/chat-message";
import type { ThreadEntry } from "./chat-thread-context";

/** One scroll owner. Reading history pauses following until the viewport returns to latest. */
export function useThreadScroll(
  entries: ThreadEntry[],
  revision: unknown,
  loadingOlder: boolean,
) {
  const viewportRef = useRef<HTMLDivElement>(null);
  const contentRef = useRef<HTMLDivElement>(null);
  const [viewportNode, setViewportNode] = useState<HTMLDivElement | null>(null);
  const [contentNode, setContentNode] = useState<HTMLDivElement | null>(null);
  const bindViewport = useCallback((node: HTMLDivElement | null) => {
    viewportRef.current = node;
    setViewportNode(node);
  }, []);
  const bindContent = useCallback((node: HTMLDivElement | null) => {
    contentRef.current = node;
    setContentNode(node);
  }, []);
  const spacerRef = useRef<HTMLDivElement>(null);
  const entriesRef = useRef(entries);
  entriesRef.current = entries;
  const mode = useRef<"bottom" | "turn" | "paused">("bottom");
  const pinnedId = useRef<string | null>(null);
  const pinnedSeen = useRef(false);
  const anchor = useRef<{ id: string; offset: number } | null>(null);
  const loadingRef = useRef(loadingOlder);
  loadingRef.current = loadingOlder;
  const [away, setAway] = useState(false);
  const [activeId, setActiveId] = useState<string | null>(null);

  const updateActive = useCallback(() => {
    const viewport = viewportRef.current;
    if (!viewport) return;
    const users = entriesRef.current.filter((entry) => entry.role === "user");
    const current =
      users.findLast(
        (entry) => entry.element.offsetTop <= viewport.scrollTop + 64,
      ) ?? users[0];
    setActiveId(current?.id ?? null);
  }, []);

  const captureAnchor = useCallback(() => {
    const viewport = viewportRef.current;
    if (!viewport) return;
    const visible = entriesRef.current.find(
      (entry) =>
        entry.element.offsetTop + entry.element.offsetHeight >
        viewport.scrollTop,
    );
    if (visible)
      anchor.current = {
        id: visible.id,
        offset: visible.element.offsetTop - viewport.scrollTop,
      };
  }, []);

  const writeTop = useCallback(
    (top: number) => {
      const viewport = viewportRef.current;
      if (!viewport) return;
      viewport.scrollTop = Math.max(0, top);
      updateActive();
    },
    [updateActive],
  );

  const follow = useCallback(() => {
    const viewport = viewportRef.current;
    const content = contentRef.current;
    const spacer = spacerRef.current;
    if (
      !viewport ||
      !content ||
      !spacer ||
      loadingRef.current ||
      anchor.current
    )
      return;
    const target = entriesRef.current.find(
      (entry) => entry.id === pinnedId.current,
    );
    if (mode.current === "turn" && target) {
      const naturalHeight = content.scrollHeight - spacer.offsetHeight;
      const trailing = Math.max(
        0,
        viewport.clientHeight - (naturalHeight - target.element.offsetTop),
      );
      if (Math.abs(spacer.offsetHeight - trailing) > 1)
        spacer.style.height = `${trailing}px`;
      writeTop(
        Math.max(
          target.element.offsetTop,
          naturalHeight - viewport.clientHeight,
        ),
      );
    } else if (mode.current === "bottom") {
      writeTop(viewport.scrollHeight);
    }
  }, [writeTop]);

  const pin = useCallback(
    (id: string) => {
      pinnedId.current = id;
      pinnedSeen.current = entriesRef.current.some((entry) => entry.id === id);
      mode.current = "turn";
      setAway(false);
      follow();
    },
    [follow],
  );

  const latest = useCallback(() => {
    anchor.current = null;
    mode.current = "bottom";
    setAway(false);
    // Existing turn space is retained so finishing a short response never jumps.
    writeTop(viewportRef.current?.scrollHeight ?? 0);
  }, [writeTop]);

  const jump = useCallback(
    (id: string) => {
      const entry = entriesRef.current.find((item) => item.id === id);
      if (!entry) return;
      mode.current = "paused";
      setAway(true);
      writeTop(entry.element.offsetTop);
      entry.element.focus({ preventScroll: true });
    },
    [writeTop],
  );

  const onScroll = useCallback(() => {
    const viewport = viewportRef.current;
    if (!viewport) return;
    const atBottom =
      viewport.scrollHeight - viewport.clientHeight - viewport.scrollTop <
      CHAT_SCROLL_FOLLOW_THRESHOLD;
    if (!atBottom) mode.current = "paused";
    else if (mode.current === "paused") mode.current = "bottom";
    setAway(!atBottom);
    if (anchor.current && mode.current === "paused") captureAnchor();
    updateActive();
  }, [captureAnchor, updateActive]);

  const onUserIntent = useCallback(() => {
    const viewport = viewportRef.current;
    if (
      viewport &&
      viewport.scrollHeight - viewport.clientHeight - viewport.scrollTop >=
        CHAT_SCROLL_FOLLOW_THRESHOLD
    ) {
      mode.current = "paused";
      setAway(true);
    }
  }, []);

  useLayoutEffect(() => {
    if (
      pinnedId.current &&
      entries.some((entry) => entry.id === pinnedId.current)
    )
      pinnedSeen.current = true;
    if (
      pinnedId.current &&
      pinnedSeen.current &&
      !entries.some((entry) => entry.id === pinnedId.current)
    ) {
      pinnedId.current = null;
      pinnedSeen.current = false;
      mode.current = "bottom";
      if (spacerRef.current) spacerRef.current.style.height = "0px";
    }
    if (anchor.current && !loadingOlder) {
      const saved = anchor.current;
      const entry = entries.find((item) => item.id === saved.id);
      if (entry) writeTop(entry.element.offsetTop - saved.offset);
      anchor.current = null;
      mode.current = "paused";
      setAway(true);
    } else {
      follow();
    }
    updateActive();
  }, [entries, revision, loadingOlder, follow, updateActive, writeTop]);

  useLayoutEffect(() => {
    const content = contentNode;
    const viewport = viewportNode;
    if (!content || !viewport) return;
    const observer = new ResizeObserver(follow);
    observer.observe(content);
    observer.observe(viewport);
    return () => observer.disconnect();
  }, [follow, contentNode, viewportNode]);

  return {
    viewportRef,
    contentRef,
    bindViewport,
    bindContent,
    spacerRef,
    away,
    activeId,
    onScroll,
    onUserIntent,
    pin,
    jump,
    latest,
    captureAnchor,
  };
}

export interface ChatThreadDemoState {
  messages: ChatThreadDemoMessage[];
  /** null explicitly terminates a restored branch. Missing keys choose the latest version. */
  selection: Record<string, string | null>;
}

export function getChatThreadDemoPath(
  state: ChatThreadDemoState,
): ChatThreadDemoMessage[] {
  const path: ChatThreadDemoMessage[] = [];
  const seen = new Set<string>();
  let parent: string | null = null;
  while (true) {
    const siblings: ChatThreadDemoMessage[] = state.messages.filter(
      (message) => message.parentId === parent,
    );
    const selected: string | null | undefined =
      state.selection[parent ?? "root"];
    const message: ChatThreadDemoMessage | undefined =
      selected === null
        ? undefined
        : selected
          ? siblings.find((item) => item.id === selected)
          : siblings.at(-1);
    if (!message || seen.has(message.id)) return path;
    path.push(message);
    seen.add(message.id);
    parent = message.id;
  }
}
