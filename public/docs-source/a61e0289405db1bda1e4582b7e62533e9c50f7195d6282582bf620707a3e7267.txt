export type MonitoringAlertSeverity = "critical" | "warning" | "general";
export type MonitoringAlertResolutionState = "pending" | "resolved";

export interface MonitoringAlertItem {
  id: string;
  resource: string;
  resourceType: string;
  environment: string;
  location: string;
  severity: MonitoringAlertSeverity;
  level: "P0" | "P1" | "P2";
  title: string;
  description: string;
  duration: string;
  state: string;
  related: string;
  resolutionState?: MonitoringAlertResolutionState;
  resolutionRecords?: readonly MonitoringAlertResolutionRecord[];
}

export interface MonitoringAlertResolutionRecord {
  id: string;
  operator: string;
  occurredAt: string;
  detail: string;
}

export interface MonitoringAlertListProps extends Omit<ComponentPropsWithoutRef<"div">, "children"> {
  workspace?: OperationsWorkspaceOptions;
  alerts?: readonly MonitoringAlertItem[];
  state?: "ready" | "loading" | "stale" | "error";
  refreshing?: boolean;
  retainDataOnError?: boolean;
  statusMessage?: string;
  onRefresh?: () => void;
  onRetry?: () => void | Promise<void>;
  onAnalyze?: (alert: MonitoringAlertItem) => void;
  onResolve?: (alert: MonitoringAlertItem) => void;
  onMute?: (alert: MonitoringAlertItem) => void;
}

function SeverityFilter({ active, count, label, level, onClick }: { active: boolean; count: number; label: string; level?: "P0" | "P1" | "P2"; onClick: () => void }) {
  const tone: BadgeStatus = level === "P0" ? "danger" : level === "P1" ? "warning" : "info";
  return <Button aria-pressed={active} className="shrink-0" onClick={onClick} size="md" type="button" variant={active ? "neutral" : "tertiary"}><span className="inline-flex items-center gap-1.5">{level ? <Badge variant="plain" status={tone} aria-label={`优先级 ${level}，${label}`}>{label}</Badge> : <span>{label}</span>}{label !== "全部" && <span className="rounded bg-surface-base px-1 text-label text-fg-muted">{count}</span>}</span></Button>;
}

function ResolutionRecords({ records }: { records: readonly MonitoringAlertResolutionRecord[] }) {
  return <div className="space-y-2"><p className="text-label font-medium text-fg-default">处置记录</p><div className="space-y-2">{records.map((record) => <div key={record.id} className="space-y-0.5 border-l-2 border-brand/30 pl-2"><p className="text-label text-fg-muted">{record.operator} 在 {record.occurredAt} 做了如下处置</p><p className="text-label text-fg-default">{record.detail}</p></div>)}</div></div>;
}

function MonitoringAlertResolutionAction({ alert, onResolve }: { alert: MonitoringAlertItem; onResolve?: (alert: MonitoringAlertItem) => void }) {
  const Check = useIcon("check-square");
  const records = alert.resolutionRecords ?? [];
  const isResolved = alert.resolutionState === "resolved";
  const label = isResolved ? "已处置" : records.length ? `处置${records.length}次` : "处置";
  const button = <Button aria-label={isResolved ? `${alert.resource} 已处置，查看处置记录` : undefined} className={cn(isResolved && "text-fg-success hover:text-fg-success")} type="button" size="sm" variant="tertiary" onClick={() => { if (!isResolved) onResolve?.(alert); }}>{isResolved ? <Badge variant="plain" status="success" size="sm" leadingIcon={<Check />}>{label}</Badge> : label}</Button>;
  if (!records.length) return button;
  return <Popover trigger="hover" hoverDelay={120} closeDelay={160}><PopoverTrigger render={button} /><PopoverContent align="end" side="bottom" sideOffset={6} className="w-80 p-3"><ResolutionRecords records={records} /></PopoverContent></Popover>;
}

function MonitoringAlertActionSplitButton({ alert, onAnalyze, onMute, onResolve }: { alert: MonitoringAlertItem; onAnalyze?: (alert: MonitoringAlertItem) => void; onMute?: (alert: MonitoringAlertItem) => void; onResolve?: (alert: MonitoringAlertItem) => void }) {
  const ChevronDown = useIcon("chevron-down");
  return <ButtonGroup aria-label={`${alert.resource} 操作`}><Button type="button" size="sm" variant="tertiary" onClick={() => onAnalyze?.(alert)}>询问 AI</Button><ButtonGroupSeparator /><DropdownMenu><DropdownTrigger render={<Button type="button" size="sm" variant="tertiary" iconOnly aria-label={`${alert.resource} 更多操作`}><ChevronDown /></Button>} /><DropdownContent align="end"><MenuItem disabled={alert.resolutionState === "resolved"} index={0} label="处置" onSelect={() => onResolve?.(alert)} /><MenuItem index={1} label="不再提醒" onSelect={() => onMute?.(alert)} /></DropdownContent></DropdownMenu></ButtonGroup>;
}

function MonitoringAlertRow({ alert, onAnalyze, onMute, onResolve }: { alert: MonitoringAlertItem; onAnalyze?: (alert: MonitoringAlertItem) => void; onMute?: (alert: MonitoringAlertItem) => void; onResolve?: (alert: MonitoringAlertItem) => void }) {
  const Monitor = useIcon("monitor");
  const severityTone = alert.severity === "critical" ? "danger" : alert.severity === "warning" ? "warning" : "info";
  return <div className="grid min-h-16 min-w-[1240px] grid-cols-[minmax(220px,1fr)_minmax(240px,1.2fr)_minmax(140px,.65fr)_minmax(180px,.85fr)_minmax(130px,.6fr)_120px_144px] items-center gap-x-3 border-b-hairline border-border px-3 py-2 last:border-b-0"><div className="flex min-w-0 items-center gap-3"><Badge status={severityTone} size="sm" variant="strong">{alert.level}</Badge><span className="grid size-8 shrink-0 place-items-center rounded-md bg-info-surface/60 text-fg-default"><Monitor aria-hidden size={20} strokeWidth={1.5} /></span><div className="min-w-0"><p className="truncate text-body font-medium text-fg-default">{alert.resource}</p><p className="truncate text-label text-fg-muted">{alert.resourceType}</p></div></div><div className="min-w-0"><p className="truncate text-body font-medium text-fg-default">{alert.title}</p><p className="truncate text-label text-fg-muted">{alert.description}</p></div><p className="truncate text-body text-fg-default">{alert.location}</p><p className="truncate text-body text-fg-default">{alert.duration}{alert.state && <span className="ms-1 text-fg-muted">{alert.state}</span>}</p><p className="whitespace-nowrap text-body text-fg-default">{alert.related}</p><div className="justify-self-end"><MonitoringAlertResolutionAction alert={alert} onResolve={onResolve} /></div><div className="justify-self-end"><MonitoringAlertActionSplitButton alert={alert} onAnalyze={onAnalyze} onMute={onMute} onResolve={onResolve} /></div></div>;
}

/** A ZAIops monitoring alert workspace with shared operations navigation and severity filters. */
export function MonitoringAlertList({ workspace, alerts = defaultMonitoringAlertItems, className, state = "ready", refreshing = false, retainDataOnError = false, statusMessage, onRefresh, onRetry, onAnalyze, onMute, onResolve, ...props }: MonitoringAlertListProps) {
  const [query, setQuery] = useState("");
  const [severity, setSeverity] = useState<MonitoringAlertSeverity | "all">("all");
  const [environment, setEnvironment] = useState("all");
  const [page, setPage] = useState(0);
  const [pageSize, setPageSize] = useState(5);
  const Search = useIcon("search");
  const Bell = useIcon("bell");
  const Monitor = useIcon("monitor");
  const Refresh = useIcon("rotate-ccw");
  const [pending, setPending] = useState<"refresh" | "retry" | null>(null);
  const inFlight = useRef(false);
  const [actionError, setActionError] = useState<{ data: typeof alerts; message: string } | null>(null);
  const currentError = actionError?.data === alerts ? actionError.message : undefined;
  const loading = state === "loading";
  const busy = loading || refreshing || pending !== null;
  const failed = state === "error" || currentError !== undefined;
  const replacesData = state === "error" && !retainDataOnError;
  async function runAction(kind: "refresh" | "retry") {
    const callback = kind === "refresh" ? onRefresh : onRetry;
    if (!callback || inFlight.current || loading || refreshing) return;
    inFlight.current = true;
    setPending(kind);
    setActionError(null);
    try { await callback(); }
    catch (cause) { setActionError({ data: alerts, message: cause instanceof Error && cause.message ? cause.message : "监控告警加载失败" }); }
    finally { inFlight.current = false; setPending(null); }
  }
  const retryAction = onRetry ? <Button disabled={busy} loading={pending === "retry"} variant="secondary" onClick={() => void runAction("retry")}>重试</Button> : undefined;
  const feedback = loading ? <div role="status" aria-label="正在加载监控告警" className="space-y-3 p-3"><Skeleton className="h-8 w-40" /><Skeleton className="h-48 w-full" /></div>
    : replacesData ? <div className="flex min-w-0 w-full items-center justify-center min-h-[min(60vh,36rem)] px-6 py-12 sm:px-10 sm:py-16"><Alert status="danger" role={Boolean(currentError) ? "alert" : "group"} className="w-full max-w-xl"><AlertTitle>{"监控告警加载失败"}</AlertTitle><AlertDescription>{currentError ?? statusMessage}</AlertDescription><AlertAction>{retryAction}</AlertAction></Alert></div>
    : <>
      {failed ? <InlineNotice tone="danger" variant="emphasized" role={currentError ? "alert" : undefined}><InlineNoticeContent>监控告警刷新失败：{currentError ?? statusMessage ?? "请稍后重试。"} 当前显示上次获取的结果。</InlineNoticeContent>{retryAction && <InlineNoticeAction>{retryAction}</InlineNoticeAction>}</InlineNotice>
        : state === "stale" && <InlineNotice tone="warning" variant="emphasized"><InlineNoticeContent>{statusMessage ?? "告警数据已过期，当前显示上次获取的结果。"}</InlineNoticeContent></InlineNotice>}
      {busy && <InlineNotice tone="info" variant="emphasized"><span aria-hidden="true" className="inline-flex h-5 shrink-0 items-center [&_svg]:size-4"><span className="inline-flex animate-spin motion-reduce:animate-none"><Refresh /></span></span><InlineNoticeContent>{"正在刷新监控告警，保留上次获取的结果。"}</InlineNoticeContent></InlineNotice>}
    </>;
  useEffect(() => setPage(0), [environment, pageSize, query, severity]);
  const visibleAlerts = useMemo(() => alerts.filter((alert) => (severity === "all" || alert.severity === severity) && (environment === "all" || alert.environment === environment) && `${alert.resource} ${alert.title} ${alert.description} ${alert.location}`.toLocaleLowerCase().includes(query.trim().toLocaleLowerCase())), [alerts, environment, query, severity]);
  useEffect(() => setPage((current) => Math.min(current, Math.max(0, Math.ceil(visibleAlerts.length / pageSize) - 1))), [visibleAlerts.length, pageSize]);
  const currentPage = Math.min(page, Math.max(0, Math.ceil(visibleAlerts.length / pageSize) - 1));
  const pagedAlerts = visibleAlerts.slice(currentPage * pageSize, (currentPage + 1) * pageSize);
  const environments = [...new Set(alerts.map((alert) => alert.environment))];
  const counts = { critical: alerts.filter((alert) => alert.severity === "critical").length, warning: alerts.filter((alert) => alert.severity === "warning").length, general: alerts.filter((alert) => alert.severity === "general").length };
  return <OperationsWorkspaceShell workspace={workspace} activeNavigation="alerts" title="监控告警" icon={Bell} className={cn("min-h-[42rem]", className)} {...props} aria-busy={busy || undefined}><PageContent><PageBody className="max-w-[101.25rem] p-3"><section className="overflow-hidden rounded-xl border-hairline border-border bg-surface-floating"><div className="flex flex-wrap items-center gap-2 border-b-hairline border-border px-3 py-2"><div className="flex flex-wrap items-center gap-1"><SeverityFilter active={severity === "all"} count={alerts.length} label="全部" onClick={() => setSeverity("all")} /><SeverityFilter active={severity === "critical"} count={counts.critical} label="严重" level="P0" onClick={() => setSeverity("critical")} /><SeverityFilter active={severity === "warning"} count={counts.warning} label="告警" level="P1" onClick={() => setSeverity("warning")} /><SeverityFilter active={severity === "general"} count={counts.general} label="一般" level="P2" onClick={() => setSeverity("general")} /></div><label className="relative min-w-52 flex-1 lg:max-w-[28rem]"><span className="sr-only">搜索监控告警</span><Search className="pointer-events-none absolute left-2 top-1/2 size-4 -translate-y-1/2 text-fg-subtle" /><Input className="pl-8" onChange={(event) => setQuery(event.target.value)} placeholder="搜索" value={query} /></label><DropdownMenu><DropdownTrigger render={<Button className="shrink-0 whitespace-nowrap" leadingIcon={Monitor} size="md" type="button" variant="tertiary">{environment === "all" ? "全部环境" : environment}</Button>} /><DropdownContent checkedIndex={environment === "all" ? 0 : environments.includes(environment) ? environments.indexOf(environment) + 1 : undefined}><MenuItem checked={environment === "all"} index={0} label="全部环境" onSelect={() => setEnvironment("all")} />{environments.map((value, index) => <MenuItem checked={environment === value} index={index + 1} key={value} label={value} onSelect={() => setEnvironment(value)} />)}</DropdownContent></DropdownMenu><Button className="ms-auto" disabled={busy} loading={refreshing || pending === "refresh"} onClick={() => void runAction("refresh")} size="md" type="button" variant="tertiary">刷新</Button></div>{feedback}{!loading && !replacesData && <><div aria-label="监控告警列表" className="overflow-x-auto">{pagedAlerts.map((alert) => <MonitoringAlertRow alert={alert} key={alert.id} onAnalyze={onAnalyze} onMute={onMute} onResolve={onResolve} />)}{!visibleAlerts.length && <Empty reason={alerts.length ? "no-filter-results" : "no-data"} scope="section"><EmptyHeader><EmptyTitle>{alerts.length ? "没有符合条件的监控告警" : "暂无监控告警"}</EmptyTitle><EmptyDescription>{alerts.length ? "调整严重程度、环境或搜索条件后重试。" : "收到监控告警后将在这里显示。"}</EmptyDescription></EmptyHeader>{alerts.length > 0 && <EmptyActions><Button variant="secondary" onClick={() => { setQuery(""); setSeverity("all"); setEnvironment("all"); setPage(0); }}>清空筛选</Button></EmptyActions>}</Empty>}</div><ListPagination onPageChange={setPage} onPageSizeChange={(nextPageSize) => { setPage(0); setPageSize(nextPageSize); }} page={currentPage} pageSize={pageSize} total={visibleAlerts.length} /></>}</section></PageBody></PageContent></OperationsWorkspaceShell>;
}
