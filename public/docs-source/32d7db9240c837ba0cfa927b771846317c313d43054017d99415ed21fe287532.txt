import type { WebsiteAnalyticsContext, WebsiteAnalyticsData, WebsiteAnalyticsLabels, WebsiteAnalyticsMetric, WebsiteAnalyticsRange, WebsiteAnalyticsRanking } from "./website-analytics-types";

export interface WebsiteAnalyticsProps {
  data: WebsiteAnalyticsData | null;
  site: string;
  range: WebsiteAnalyticsRange;
  onRangeChange?: (range: WebsiteAnalyticsRange) => void;
  metric?: WebsiteAnalyticsMetric;
  onMetricChange?: (metric: WebsiteAnalyticsMetric) => void;
  compare?: boolean;
  onCompareChange?: (compare: boolean) => void;
  onExport?: (context: WebsiteAnalyticsContext) => void;
  exporting?: boolean;
  state?: "ready" | "loading" | "error";
  onRetry?: () => void;
  labels?: Partial<WebsiteAnalyticsLabels>;
  locale?: string;
  timeZone?: string;
  className?: string;
}

const metrics: WebsiteAnalyticsMetric[] = ["visitors", "signups", "conversion", "bounceRate"];
const ranges: WebsiteAnalyticsRange[] = ["7d", "30d", "90d"];
const currentColor = chartCssVars.linePrimary;
const previousColor = chartCssVars.lineSecondary;
const isRate = (metric: WebsiteAnalyticsMetric) => metric === "conversion" || metric === "bounceRate";
const finite = (value: unknown): value is number => typeof value === "number" && Number.isFinite(value);
const metricValue = (value: unknown, metric: WebsiteAnalyticsMetric) => finite(value) && value >= 0 && (!isRate(metric) || value <= 1) ? value : null;

function RankingLabels({ items, formatCount }: { items: WebsiteAnalyticsRanking[]; formatCount: (value: unknown) => string }) {
  const { barScale, bandWidth, yScale, innerWidth, margin } = useChartStable();
  if (!barScale || !bandWidth || innerWidth <= 0) return null;
  return <g className="pointer-events-none">
    {items.map((item) => {
      const barEnd = finite(item.visitors) && item.visitors >= 0 ? yScale(item.visitors) : 0;
      const y = barScale(item.id) ?? 0;
      const count = formatCount(item.visitors);
      return <g key={item.id}>
        <foreignObject x={0} y={y} width={innerWidth} height={bandWidth}>
          <div className="relative flex h-full items-center px-3 text-body">
            <span className="pointer-events-auto min-w-0 truncate text-fg-default" title={item.label}>{item.label}</span>
            <div aria-hidden="true" className="absolute inset-0 flex items-center px-3 text-fg-on-inverse" style={{ clipPath: `inset(0 ${Math.max(0, innerWidth - barEnd)}px 0 0)` }}>
              <span className="min-w-0 truncate">{item.label}</span>
            </div>
          </div>
        </foreignObject>
        <foreignObject x={innerWidth} y={y} width={margin.right} height={bandWidth}>
          <div className="flex h-full items-center justify-end pr-3 text-body tabular-nums text-fg-muted"><span className="pointer-events-auto truncate" title={count}>{count}</span></div>
        </foreignObject>
      </g>;
    })}
  </g>;
}

function Rankings({ title, items, labels, formatCount }: { title: string; items: WebsiteAnalyticsRanking[]; labels: WebsiteAnalyticsLabels; formatCount: (value: unknown) => string }) {
  const titleId = useId();
  const ranked = [...items].sort((a, b) => (finite(b.visitors) && b.visitors >= 0 ? b.visitors : -1) - (finite(a.visitors) && a.visitors >= 0 ? a.visitors : -1));
  return <section className="min-w-0 p-5" aria-labelledby={titleId}>
    <div className="mb-3 flex items-baseline justify-between gap-3">
      <h3 id={titleId} className="text-body font-medium text-fg-default">{title}</h3>
      <span className="text-label text-fg-subtle">{labels.visitors}</span>
    </div>
    {ranked.length === 0 ? <p className="py-8 text-body text-fg-muted">{labels.empty}</p> : <div style={{ height: `calc(var(--spacing) * ${ranked.length * 11})` }}>
      <BarChart data={ranked.map((item) => ({ id: item.id, label: item.label, visitors: finite(item.visitors) && item.visitors >= 0 ? item.visitors : null }))} xDataKey="id" orientation="horizontal" aspectRatio="auto" className="h-full" margin={{ top: 0, right: 80, bottom: 0, left: 0 }} barGap={0.12}>
        <Bar dataKey="visitors" lineCap={8} fadedOpacity={1} animate={false} />
        <RankingLabels items={ranked} formatCount={formatCount} />
      </BarChart>
      <ol className="sr-only">{ranked.map((item) => <li key={item.id}>{item.label}: {formatCount(item.visitors)}</li>)}</ol>
    </div>}
  </section>;
}

/** An embeddable website-analytics surface. Queries, period alignment and exports belong to the host. */
export function WebsiteAnalytics({ data, site, range, onRangeChange, metric: controlledMetric, onMetricChange, compare: controlledCompare, onCompareChange, onExport, exporting = false, state = "ready", onRetry, labels: overrides, locale = "en-US", timeZone = "UTC", className }: WebsiteAnalyticsProps) {
  const titleId = useId();
  const chartId = useId();
  const compareId = useId();
  const labels = { ...websiteAnalyticsEnglishLabels, ...overrides };
  const [localMetric, setLocalMetric] = useState<WebsiteAnalyticsMetric>("visitors");
  const [localCompare, setLocalCompare] = useState(false);
  const metric = controlledMetric ?? localMetric;
  const requestedCompare = controlledCompare ?? localCompare;
  const ArrowUp = useIcon("arrow-up");
  const ArrowDown = useIcon("arrow-down");
  const Check = useIcon("check");
  const snapshot = data?.range === range && state === "ready" ? data : null;
  const pending = state === "loading" || (state === "ready" && data !== null && data.range !== range);
  const { number, percent, points, compact, date } = useMemo(() => ({
    number: new Intl.NumberFormat(locale, { maximumFractionDigits: 0 }),
    percent: new Intl.NumberFormat(locale, { style: "percent", maximumFractionDigits: 1 }),
    points: new Intl.NumberFormat(locale, { maximumFractionDigits: 1 }),
    compact: new Intl.NumberFormat(locale, { notation: "compact", maximumFractionDigits: 1 }),
    date: new Intl.DateTimeFormat(locale, { month: "short", day: "numeric", timeZone }),
  }), [locale, timeZone]);
  const formatCount = (value: unknown) => finite(value) && value >= 0 ? number.format(value) : "—";
  const formatValue = (value: unknown, key: WebsiteAnalyticsMetric = metric) => {
    const valid = metricValue(value, key);
    return valid === null ? "—" : isRate(key) ? percent.format(valid) : number.format(valid);
  };
  const formatDate = (value: string | Date) => {
    const parsed = new Date(value);
    return Number.isFinite(parsed.getTime()) ? date.format(parsed) : "—";
  };
  const trend = useMemo(() => (snapshot?.trend ?? [])
    .filter((point) => Number.isFinite(new Date(point.date).getTime()))
    .map((point) => ({ date: point.date, current: metricValue(point[metric], metric), previous: metricValue(point.comparison?.[metric], metric) })), [snapshot, metric]);
  const hasTrend = trend.some((point) => point.current !== null);
  const hasComparison = trend.some((point) => point.previous !== null);
  const compare = requestedCompare && hasComparison;
  const context: WebsiteAnalyticsContext = { range, metric, compare };
  const rangeLabel = labels[range === "7d" ? "range7d" : range === "90d" ? "range90d" : "range30d"];

  return <Container className={cn("w-full", className)} role="region" aria-labelledby={titleId} data-slot="website-analytics" aria-busy={pending || undefined}>
    <ContainerHeader className="py-2">
      <div className="min-w-0 flex-1">
        <h2 id={titleId} className="text-body font-medium text-fg-default">{labels.title}</h2>
        <p className="mt-1 break-words text-label text-fg-muted">{site} · {rangeLabel}</p>
      </div>
      <div className="flex w-full max-w-full flex-wrap items-center justify-between gap-2 sm:w-auto sm:justify-end">
        <Tabs value={range} onValueChange={(value) => onRangeChange?.(value as WebsiteAnalyticsRange)} variant="segment" color="default">
          <TabsList aria-label={labels.range}>{ranges.map((value) => <TabItem key={value} value={value} disabled={!onRangeChange} label={value.toUpperCase()} />)}</TabsList>
        </Tabs>
        {onExport && <Button variant="secondary" size="sm" leadingIcon={ArrowDown} onClick={() => onExport(context)} loading={exporting} disabled={!snapshot}>{labels.export}</Button>}
      </div>
    </ContainerHeader>
    <ContainerBody className="p-0">
      <div className="grid grid-cols-2 gap-3 p-4 lg:grid-cols-4">
        {metrics.map((key) => {
          const summary = snapshot?.metrics[key];
          const delta = summary?.change;
          const positive = finite(delta) && (key === "bounceRate" ? delta < 0 : delta > 0);
          const DeltaIcon = finite(delta) && delta < 0 ? ArrowDown : ArrowUp;
          return <MetricCard key={key} label={labels[key]} value={formatValue(summary?.value, key)} state={pending ? "loading" : state === "error" ? "error" : snapshot ? "ready" : "unavailable"}
            onClick={() => { if (controlledMetric === undefined) setLocalMetric(key); onMetricChange?.(key); }} actionLabel={`${labels[key]}${metric === key ? ` · ${labels.selected}` : ""}`}
            leading={metric === key ? <Check aria-label={labels.selected} className="size-4 text-fg-default" /> : undefined}
            footer={<span className={cn("flex items-center gap-1 text-label tabular-nums", !finite(delta) || delta === 0 ? "text-fg-muted" : positive ? "text-fg-success" : "text-fg-danger")}>
              {finite(delta) && delta !== 0 && <DeltaIcon aria-hidden="true" className="size-3.5" />}
              {finite(delta) ? isRate(key) ? `${points.format(Math.abs(delta) * 100)} ${labels.points}` : percent.format(Math.abs(delta)) : "—"}
            </span>} />;
        })}
      </div>
      {state === "error" ? <div className="flex flex-wrap items-center gap-3 p-5"><InlineNotice variant="emphasized" tone="danger" role="alert">{labels.error}</InlineNotice>{onRetry && <Button variant="secondary" size="sm" onClick={onRetry}>{labels.retry}</Button>}</div> : <>
        <section className="border-t-hairline border-border p-5" aria-labelledby={chartId}>
          <div className="mb-3 flex flex-wrap items-center justify-between gap-3">
            <h3 id={chartId} className="sr-only">{labels[metric]}</h3>
            <ChartLegend className="flex flex-wrap" items={[{ id: "current", label: labels[metric], color: currentColor }, ...(compare ? [{ id: "previous", label: labels.previousPeriod, color: previousColor }] : [])]} />
            <label htmlFor={compareId} className="flex cursor-pointer items-center gap-2 text-label text-fg-muted">
              <Checkbox id={compareId} checked={compare} disabled={!hasComparison || pending} onCheckedChange={(value) => { if (controlledCompare === undefined) setLocalCompare(value); onCompareChange?.(value); }} />{labels.compare}
            </label>
          </div>
          {pending || hasTrend ? <AreaChart data={trend} status={pending ? "loading" : "ready"} loadingLabel={labels.loading} style={{ height: "calc(var(--spacing) * 72)", aspectRatio: "auto" }} margin={{ top: 12, right: 8, bottom: 32, left: 48 }}>
            <Grid horizontal numTicksRows={4} />
            <Area dataKey="current" />
            {compare && <Area dataKey="previous" fill={previousColor} fillOpacity={0} dashFromIndex={0} dashArray="4,4" showHighlight={false} />}
            <XAxis numTicks={6} formatDate={(value) => formatDate(value)} />
            <YAxis numTicks={4} formatValue={(value) => isRate(metric) ? percent.format(value) : compact.format(value)} />
            <ChartTooltip showDatePill={false} content={({ point }) => <TooltipContent title={formatDate(String(point.date))} rows={[{ color: currentColor, label: labels[metric], value: formatValue(point.current) }, ...(compare ? [{ color: previousColor, label: labels.previousPeriod, value: formatValue(point.previous) }] : [])]} />} />
          </AreaChart> : <p role="status" className="flex min-h-72 items-center justify-center text-body text-fg-muted">{labels.empty}</p>}
          {snapshot && trend.length > 0 && <ChartDataTable caption={labels[metric]} summary={labels.viewData} columns={[labels.date, labels[metric], ...(compare ? [labels.previousPeriod] : [])]} rows={trend.map((point, index) => ({ id: `${point.date}-${index}`, label: formatDate(point.date), values: [formatValue(point.current), ...(compare ? [formatValue(point.previous)] : [])] }))} />}
        </section>
        {!pending && <div className="grid border-t-hairline border-border md:grid-cols-2">
          <Rankings title={labels.topPages} items={snapshot?.pages ?? []} labels={labels} formatCount={formatCount} />
          <div className="min-w-0 border-t-hairline border-border md:border-l-hairline md:border-t-0"><Rankings title={labels.sources} items={snapshot?.sources ?? []} labels={labels} formatCount={formatCount} /></div>
        </div>}
      </>}
    </ContainerBody>
  </Container>;
}


export type WebsiteAnalyticsRange = "7d" | "30d" | "90d";
export type WebsiteAnalyticsMetric = "visitors" | "signups" | "conversion" | "bounceRate";

export interface WebsiteAnalyticsSummary {
  value: number | null;
  /** Counts: relative change as a ratio. Rates: absolute change as a ratio (0.001 = 0.1 percentage points). */
  change: number | null;
}

export interface WebsiteAnalyticsPoint {
  /** ISO date or timestamp, including the offset when a time is supplied. */
  date: string;
  visitors: number | null;
  signups: number | null;
  /** Rates are ratios in [0, 1]. */
  conversion: number | null;
  bounceRate: number | null;
  /** Previous-period buckets aligned by the host to this point. Missing values stay unknown. */
  comparison?: Partial<Record<WebsiteAnalyticsMetric, number | null>>;
}

export interface WebsiteAnalyticsRanking {
  id: string;
  label: string;
  visitors: number | null;
}

export interface WebsiteAnalyticsData {
  /** Used to avoid displaying the previous range's response under a new range label. */
  range: WebsiteAnalyticsRange;
  metrics: Record<WebsiteAnalyticsMetric, WebsiteAnalyticsSummary>;
  trend: WebsiteAnalyticsPoint[];
  pages: WebsiteAnalyticsRanking[];
  sources: WebsiteAnalyticsRanking[];
}

export interface WebsiteAnalyticsLabels {
  title: string;
  visitors: string;
  signups: string;
  conversion: string;
  bounceRate: string;
  range: string;
  range7d: string;
  range30d: string;
  range90d: string;
  compare: string;
  previousPeriod: string;
  export: string;
  topPages: string;
  sources: string;
  date: string;
  viewData: string;
  points: string;
  selected: string;
  empty: string;
  loading: string;
  error: string;
  retry: string;
}

export interface WebsiteAnalyticsContext {
  range: WebsiteAnalyticsRange;
  metric: WebsiteAnalyticsMetric;
  compare: boolean;
}

import type { WebsiteAnalyticsData, WebsiteAnalyticsPoint, WebsiteAnalyticsRange } from "./website-analytics-types";

const dailyVisitors = [1130, 1135, 1220, 1260, 1330, 1070, 1010, 1250, 1290, 1310, 1410, 1450, 1090, 1060, 1330, 1280, 1340, 1410, 1410, 1100, 1060, 1240, 1240, 1310, 1340, 1380, 1090, 1010, 1250, 1280];

/** Fixed, illustrative data. No analytics service is queried. */
export function createWebsiteAnalyticsDemoData(range: WebsiteAnalyticsRange = "30d"): WebsiteAnalyticsData {
  const days = range === "7d" ? 7 : range === "90d" ? 90 : 30;
  const scale = 36686 / dailyVisitors.reduce((sum, value) => sum + value, 0);
  const month = dailyVisitors.map((value) => Math.round(value * scale));
  month[month.length - 1] += 36686 - month.reduce((sum, value) => sum + value, 0);
  const registrations = month.map((value) => Math.round(value * 1746 / 36686));
  registrations[registrations.length - 1] += 1746 - registrations.reduce((sum, value) => sum + value, 0);
  const priorVisitors = month.map((value) => Math.round(value / 1.117));
  const priorSignups = registrations.map((value) => Math.round(value / 1.151));
  priorVisitors[29] += Math.round(36686 / 1.117) - priorVisitors.reduce((sum, value) => sum + value, 0);
  priorSignups[29] += Math.round(1746 / 1.151) - priorSignups.reduce((sum, value) => sum + value, 0);
  const trend: WebsiteAnalyticsPoint[] = Array.from({ length: days }, (_, index) => {
    const day = days === 7 ? 23 + index : index % 30;
    const visitors = month[day];
    const signups = registrations[day];
    const previousVisitors = priorVisitors[day];
    const previousSignups = priorSignups[day];
    return {
      date: new Date(Date.UTC(2026, 8, 30 - days + 1 + index)).toISOString(),
      visitors, signups, conversion: signups / visitors, bounceRate: 0.406,
      comparison: { visitors: previousVisitors, signups: previousSignups, conversion: previousSignups / previousVisitors, bounceRate: 0.413 },
    };
  });
  const visitors = trend.reduce((sum, point) => sum + (point.visitors ?? 0), 0);
  const signups = trend.reduce((sum, point) => sum + (point.signups ?? 0), 0);
  const previousVisitors = trend.reduce((sum, point) => sum + (point.comparison?.visitors ?? 0), 0);
  const previousSignups = trend.reduce((sum, point) => sum + (point.comparison?.signups ?? 0), 0);
  const multiplier = visitors / 36686;
  return {
    range, trend,
    metrics: {
      visitors: { value: visitors, change: visitors / previousVisitors - 1 },
      signups: { value: signups, change: signups / previousSignups - 1 },
      conversion: { value: signups / visitors, change: signups / visitors - previousSignups / previousVisitors },
      bounceRate: { value: 0.406, change: -0.007 },
    },
    pages: [
      { id: "home", label: "/", visitors: Math.round(11373 * multiplier) },
      { id: "pricing", label: "/pricing", visitors: Math.round(6237 * multiplier) },
      { id: "docs", label: "/docs/getting-started", visitors: Math.round(4402 * multiplier) },
      { id: "blog", label: "/blog/usage-based-billing", visitors: Math.round(2935 * multiplier) },
      { id: "changelog", label: "/changelog", visitors: Math.round(1834 * multiplier) },
    ],
    sources: [
      { id: "direct", label: "Direct", visitors: Math.round(13941 * multiplier) },
      { id: "google", label: "Google", visitors: Math.round(10639 * multiplier) },
      { id: "hacker-news", label: "Hacker News", visitors: Math.round(4035 * multiplier) },
      { id: "x", label: "X", visitors: Math.round(2935 * multiplier) },
      { id: "newsletter", label: "Newsletter", visitors: Math.round(2201 * multiplier) },
    ],
  };
}
