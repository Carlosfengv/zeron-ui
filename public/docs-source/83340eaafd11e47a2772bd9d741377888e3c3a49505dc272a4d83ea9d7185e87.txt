/** A ZAIops inspection report list with the shared operations navigation shell. */
export function InspectionReportList({ workspace, reports = defaultInspectionReportItems, className, onRefresh, onViewReport, ...props }: InspectionReportListProps) {
  const [query, setQuery] = useState(""); const [priority, setPriority] = useState<InspectionReportPriority | "all">("all"); const [environment, setEnvironment] = useState("all"); const [page, setPage] = useState(0); const [pageSize, setPageSize] = useState(5); const Search = useIcon("search"); const Check = useIcon("check-square");
  useEffect(() => setPage(0), [environment, pageSize, priority, query]);
  const visibleReports = useMemo(() => reports.filter((report) => (priority === "all" || report.priority === priority) && (environment === "all" || report.environment === environment) && `${report.name} ${report.location} ${report.issue ?? ""}`.toLocaleLowerCase().includes(query.trim().toLocaleLowerCase())), [environment, priority, query, reports]);
  useEffect(() => setPage(current => Math.min(current, Math.max(0, Math.ceil(visibleReports.length / pageSize) - 1))), [visibleReports.length, pageSize]);
  const currentPage = Math.min(page, Math.max(0, Math.ceil(visibleReports.length / pageSize) - 1));
  const pagedReports = visibleReports.slice(currentPage * pageSize, (currentPage + 1) * pageSize);
  const environments = [...new Set(reports.map((report) => report.environment))];
  return <OperationsWorkspaceShell workspace={workspace} activeNavigation="reports" title="巡检报告" icon={Check} className={cn("min-h-[42rem]", className)} {...props}><PageContent><PageBody className="max-w-[101.25rem] p-3"><section className="overflow-hidden rounded-xl border-hairline border-border bg-surface-floating"><div className="flex flex-wrap items-center gap-2 border-b-hairline border-border px-3 py-2"><label className="relative min-w-56 flex-1 lg:max-w-[28rem]"><span className="sr-only">搜索巡检报告</span><Search className="pointer-events-none absolute left-2 top-1/2 size-4 -translate-y-1/2 text-fg-subtle" /><Input className="pl-8" onChange={(event) => setQuery(event.target.value)} placeholder="搜索" value={query} /></label><DropdownMenu><DropdownTrigger render={<Button size="md" type="button" variant="tertiary">处置优先级</Button>} /><DropdownContent>{(["all", "critical", "normal"] as const).map((value, index) => <MenuItem checked={priority === value} index={index} key={value} label={value === "all" ? "全部优先级" : value === "critical" ? "严重 · P1" : "正常"} onSelect={() => setPriority(value)} />)}</DropdownContent></DropdownMenu><DropdownMenu><DropdownTrigger render={<Button size="md" type="button" variant="tertiary">{environment === "all" ? "全部环境" : environment}</Button>} /><DropdownContent><MenuItem checked={environment === "all"} index={0} label="全部环境" onSelect={() => setEnvironment("all")} />{environments.map((value, index) => <MenuItem checked={environment === value} index={index + 1} key={value} label={value} onSelect={() => setEnvironment(value)} />)}</DropdownContent></DropdownMenu><Button className="ms-auto" onClick={onRefresh} size="md" type="button" variant="neutral">刷新</Button></div><div className="overflow-x-auto" aria-label="巡检报告列表">{pagedReports.map((report) => <ReportRow key={report.id} onView={onViewReport} report={report} />)}{!visibleReports.length && <Empty scope="section" reason={query.trim() || priority !== "all" || environment !== "all" ? "no-results" : "no-data"}><EmptyDescription>{query.trim() || priority !== "all" || environment !== "all" ? "未找到符合条件的巡检报告。" : "暂无巡检报告。"}</EmptyDescription></Empty>}</div><ListPagination onPageChange={setPage} onPageSizeChange={(nextPageSize) => { setPage(0); setPageSize(nextPageSize); }} page={currentPage} pageSize={pageSize} total={visibleReports.length} /></section></PageBody></PageContent></OperationsWorkspaceShell>;
}
