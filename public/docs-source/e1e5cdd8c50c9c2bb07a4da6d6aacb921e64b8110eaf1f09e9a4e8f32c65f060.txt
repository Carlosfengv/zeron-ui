import type { ArtifactResource, ArtifactView } from "./chat-workspace-types";

export function artifactVersionKey(resource: ArtifactResource) {
  return JSON.stringify([resource.id, resource.version]);
}

export function artifactViews(resource: ArtifactResource): ArtifactView[] {
  if (resource.kind === "code") return resource.previousContent !== undefined ? ["diff", "source"] : ["source"];
  if (resource.kind === "markdown") return resource.previousContent !== undefined ? ["preview", "source", "diff"] : ["preview", "source"];
  return resource.kind === "text" ? ["source"] : ["preview"];
}

export function artifactIsReviewable(resource: ArtifactResource) {
  return (resource.kind === "code" || resource.kind === "markdown") && resource.previousContent !== undefined && resource.review !== undefined;
}

export function artifactReviewStatus(resource: ArtifactResource) {
  return artifactIsReviewable(resource) && "review" in resource ? resource.review : undefined;
}

/** Ready revisions awaiting a decision, preserving the host's resource order. */
export function getPendingReviewResources(resources: readonly ArtifactResource[]) {
  return resources.filter((resource) => artifactReviewStatus(resource) === "pending" && (!resource.status || resource.status === "ready"));
}

/** Web frames only receive HTTP(S). File/image URLs may also be local blob URLs. */
export function safeArtifactUrl(url?: string, web = false): string | undefined {
  if (!url?.trim()) return undefined;
  try {
    const parsed = new URL(url, "https://preview.invalid");
    return (web ? ["http:", "https:"] : ["http:", "https:", "blob:"]).includes(parsed.protocol) ? url : undefined;
  } catch {
    return undefined;
  }
}

export function artifactText(resource: ArtifactResource) {
  return "content" in resource ? resource.content : "url" in resource ? safeArtifactUrl(resource.url, resource.kind === "url") : undefined;
}

export function artifactSizeText(resource: ArtifactResource, locale = "en") {
  const bytes = resource.size !== undefined && Number.isFinite(resource.size) && resource.size >= 0
    ? resource.size : "content" in resource ? new TextEncoder().encode(resource.content).byteLength : undefined;
  if (bytes === undefined) return undefined;
  const units = ["B", "KB", "MB", "GB", "TB"];
  const unit = bytes > 0 ? Math.max(0, Math.min(Math.floor(Math.log(bytes) / Math.log(1024)), units.length - 1)) : 0;
  return `${new Intl.NumberFormat(locale, { maximumFractionDigits: unit ? 1 : 0 }).format(bytes / 1024 ** unit)} ${units[unit]}`;
}

export function downloadArtifact(resource: ArtifactResource) {
  const remote = safeArtifactUrl(resource.downloadUrl) ?? ("url" in resource ? safeArtifactUrl(resource.url, resource.kind === "url") : undefined);
  const blob = !remote && "content" in resource ? URL.createObjectURL(new Blob([resource.content], { type: "text/plain;charset=utf-8" })) : undefined;
  const url = remote ?? blob;
  if (!url) return;
  const link = document.createElement("a");
  link.href = url;
  link.download = resource.name;
  link.rel = "noopener noreferrer";
  link.target = "_blank";
  link.click();
  if (blob) setTimeout(() => URL.revokeObjectURL(blob), 1000);
}

function ImagePreview({ resource, labels }: { resource: Extract<ArtifactResource, { kind: "image" }>; labels: ArtifactLabels }) {
  const viewport = useRef<HTMLDivElement>(null);
  const [bounds, setBounds] = useState({ width: 0, height: 0 });
  const [dimensions, setDimensions] = useState({ width: 0, height: 0 });
  const [zoom, setZoom] = useState(1);
  const [failed, setFailed] = useState(false);
  const [attempt, setAttempt] = useState(0);
  const url = safeArtifactUrl(resource.url);
  useEffect(() => {
    const node = viewport.current;
    if (!node) return;
    const measure = () => setBounds({ width: node.clientWidth, height: node.clientHeight });
    measure();
    const observer = new ResizeObserver(measure);
    observer.observe(node);
    return () => observer.disconnect();
  }, []);
  const fit = dimensions.width && bounds.width ? Math.min(Math.max(1, bounds.width - 48) / dimensions.width, Math.max(1, bounds.height - 48) / dimensions.height, 1) : 1;
  return <div className="flex min-h-0 flex-1 flex-col">
    <div className="flex shrink-0 items-center justify-between gap-3 border-b-hairline border-border px-4 py-2">
      <span className="text-label text-fg-muted">{dimensions.width ? `${dimensions.width} × ${dimensions.height}` : labels.preview}</span>
      <div className="flex items-center gap-2">
        <Button size="xs" variant="ghost" iconOnly aria-label={labels.zoomOut} disabled={zoom <= 0.25 || !dimensions.width} onClick={() => setZoom((value) => Math.max(0.25, value / 1.25))}>−</Button>
        <span className="w-12 text-center text-label tabular-nums">{Math.round(fit * zoom * 100)}%</span>
        <Button size="xs" variant="ghost" iconOnly aria-label={labels.zoomIn} disabled={zoom >= 8 || !dimensions.width} onClick={() => setZoom((value) => Math.min(8, value * 1.25))}>+</Button>
        <Button size="xs" variant="ghost" onClick={() => setZoom(1)}>{labels.fit}</Button>
      </div>
    </div>
    <div ref={viewport} className="relative min-h-0 flex-1 overflow-auto bg-surface-raised" tabIndex={0} aria-label={`${labels.preview}: ${resource.name}`}>
      {!url ? <PreviewNotice>{labels.unavailableUrl}</PreviewNotice> : failed ? <PreviewNotice><span>{labels.failed} </span><Button size="sm" variant="ghost" onClick={() => { setFailed(false); setDimensions({ width: 0, height: 0 }); setAttempt((value) => value + 1); }}>{labels.retry}</Button></PreviewNotice> :
        <div className="grid min-h-full min-w-full w-max place-items-center p-6">
          {!dimensions.width && <span role="status" className="text-label text-fg-muted">{labels.loading}</span>}
          {/* eslint-disable-next-line @next/next/no-img-element -- natural dimensions drive the zoomable viewer. */}
          <img key={attempt} src={url} alt={resource.alt ?? resource.name} draggable={false} className="block max-w-none"
            style={{ width: dimensions.width ? dimensions.width * fit * zoom : undefined, height: dimensions.height ? dimensions.height * fit * zoom : undefined, visibility: dimensions.width ? "visible" : "hidden" }}
            onLoad={(event) => setDimensions({ width: event.currentTarget.naturalWidth, height: event.currentTarget.naturalHeight })} onError={() => setFailed(true)} />
        </div>}
    </div>
  </div>;
}

function WebPreview({ resource, labels }: { resource: Extract<ArtifactResource, { kind: "url" }>; labels: ArtifactLabels }) {
  const [attempt, setAttempt] = useState(0);
  const [loading, setLoading] = useState(true);
  const url = safeArtifactUrl(resource.url, true);
  return <div className="flex min-h-0 flex-1 flex-col">
    <div className="flex shrink-0 items-center gap-2 border-b-hairline border-border px-4 py-2">
      <span className="min-w-0 flex-1 truncate font-mono text-label text-fg-muted" title={url}>{url ?? labels.unavailableUrl}</span>
      <Button size="xs" variant="ghost" disabled={!url} onClick={() => { setLoading(true); setAttempt((value) => value + 1); }}>{labels.refresh}</Button>
    </div>
    <p className="shrink-0 px-4 py-2 text-label text-fg-muted">{labels.webNotice}</p>
    {url ? <div className="relative min-h-0 flex-1">
      {loading && <span role="status" className="absolute left-4 top-3 text-label text-fg-muted">{labels.loading}</span>}
      <iframe key={attempt} src={url} title={resource.name} sandbox="allow-scripts allow-forms allow-popups allow-popups-to-escape-sandbox" referrerPolicy="no-referrer" className="size-full border-0 bg-surface-floating" onLoad={() => setLoading(false)} />
    </div> : <PreviewNotice>{labels.unavailableUrl}</PreviewNotice>}
  </div>;
}

export function ArtifactPreview({ resource, view, split, labels, renderPreview, onRetry }: {
  resource: ArtifactResource; view: ArtifactView; split: boolean; labels: ArtifactLabels;
  renderPreview?: ArtifactPanelProps["renderPreview"]; onRetry?: () => void;
}) {
  if (resource.status === "loading") return <PreviewNotice><span role="status">{labels.loading}</span></PreviewNotice>;
  if (resource.status === "error") return <PreviewNotice><span role="alert">{resource.error ?? labels.failed}</span>{onRetry && <Button variant="ghost" size="sm" onClick={onRetry}>{labels.retry}</Button>}</PreviewNotice>;
  const custom = renderPreview?.(resource);
  if (custom !== undefined) return custom;
  if ("content" in resource) return <Suspense fallback={<PreviewNotice>{labels.loading}</PreviewNotice>}><CodePreview resource={resource} view={view} split={split} /></Suspense>;
  if (resource.kind === "image") return <ImagePreview key={resource.url} resource={resource} labels={labels} />;
  if (resource.kind === "url") return <WebPreview key={resource.url} resource={resource} labels={labels} />;
  return <div className="flex flex-1 items-center justify-center p-6"><Empty reason="no-data" scope="inline" density="compact">
    <FileType fileName={resource.name} size="xl" />
    <EmptyHeader><EmptyTitle>{resource.name}</EmptyTitle><EmptyDescription>{labels.unsupported}</EmptyDescription></EmptyHeader>
    <p className="text-label text-fg-muted">{[resource.mimeType, resource.size !== undefined && `${resource.size.toLocaleString()} B`].filter(Boolean).join(" · ")}</p>
  </Empty></div>;
}

export class ArtifactPreviewBoundary extends Component<{ children: ReactNode; labels: ArtifactLabels }, { failed: boolean }> {
  state = { failed: false };
  static getDerivedStateFromError() { return { failed: true }; }
  render() {
    return this.state.failed ? <PreviewNotice><span role="alert">{this.props.labels.failed}</span><Button size="sm" variant="ghost" onClick={() => this.setState({ failed: false })}>{this.props.labels.retry}</Button></PreviewNotice> : this.props.children;
  }
}

/** Keep the selected resource visible and fit complete tabs in their original order. */
export function visibleArtifactIds(resources: readonly ArtifactResource[], activeId: string, widths: readonly number[], available: number, moreWidth: number, gap: number): Set<string> {
  if (available <= 0 || widths.length !== resources.length || widths.some((width) => width <= 0)) return new Set(resources.map((resource) => resource.id));
  if (widths.reduce((sum, width) => sum + width, 0) + Math.max(0, resources.length - 1) * gap <= available) return new Set(resources.map((resource) => resource.id));
  const activeIndex = Math.max(0, resources.findIndex((resource) => resource.id === activeId));
  const visible = new Set([resources[activeIndex].id]);
  let used = widths[activeIndex];
  const budget = available - moreWidth;
  resources.forEach((resource, index) => {
    if (index !== activeIndex && used + gap + widths[index] <= budget) {
      visible.add(resource.id);
      used += gap + widths[index];
    }
  });
  return visible;
}

export function ArtifactTabs({ resources, activeId, labels, onSelect, onClose, addAction }: {
  resources: readonly ArtifactResource[];
  activeId: string;
  labels: ArtifactLabels;
  onSelect: (id: string) => void;
} & Pick<ArtifactPanelProps, "onClose" | "addAction">) {
  const availableRef = useRef<HTMLDivElement>(null);
  const measureRef = useRef<HTMLDivElement>(null);
  const moreRef = useRef<HTMLDivElement>(null);
  const [layout, setLayout] = useState({ available: 0, widths: [] as number[], moreWidth: 0, gap: 0 });
  const Globe = useIcon("globe");
  const More = useIcon("ellipsis");
  const closable = !!onClose;
  // Content and review updates do not change tab geometry (for example while a reply streams).
  const measurementKey = JSON.stringify(resources.map(({ id, name, kind }) => [id, name, kind]));
  useLayoutEffect(() => {
    const available = availableRef.current;
    const measurement = measureRef.current;
    const more = moreRef.current;
    if (!available || !measurement || !more) return;
    const measure = () => {
      const widths = Array.from(measurement.querySelectorAll<HTMLElement>("[data-measure-artifact]")).map((tab) => (tab.closest<HTMLElement>('[data-slot="closable-tab"]') ?? tab).getBoundingClientRect().width);
      const gap = parseFloat(getComputedStyle(measurement).columnGap) || 0;
      const actionGap = parseFloat(getComputedStyle(available).columnGap) || 0;
      const next = { available: available.clientWidth, widths, gap, moreWidth: more.getBoundingClientRect().width + actionGap };
      setLayout((current) => current.available === next.available && current.moreWidth === next.moreWidth && current.gap === gap && current.widths.length === widths.length && current.widths.every((width, index) => width === widths[index]) ? current : next);
    };
    measure();
    if (typeof ResizeObserver === "undefined") return;
    const observer = new ResizeObserver(measure);
    observer.observe(available);
    observer.observe(measurement);
    observer.observe(more);
    return () => observer.disconnect();
  }, [measurementKey, activeId, closable]);
  const visible = visibleArtifactIds(resources, activeId, layout.widths, layout.available, layout.moreWidth, layout.gap);
  const leading = (resource: ArtifactResource) => resource.kind === "url" ? <Globe size={16} aria-hidden /> : <FileType fileName={resource.name} variant="mono" size="mini" aria-hidden />;
  const focusSelected = (id: string) => requestAnimationFrame(() => {
    Array.from(availableRef.current?.querySelectorAll<HTMLElement>("[data-artifact-id]") ?? []).find((tab) => tab.dataset.artifactId === id)?.focus();
  });
  return <div data-slot="artifact-tab-bar" className="relative flex shrink-0 items-center gap-2 border-b-hairline border-border bg-surface-floating p-2">
    <div ref={availableRef} className="flex min-w-0 flex-1 items-center gap-2">
      <div className="min-w-0 flex-1">
        <TabsList aria-label={labels.resources}>
          {resources.filter((resource) => visible.has(resource.id)).map((resource) => <TabItem key={resource.id} value={resource.id} label={resource.name} title={resource.name} data-artifact-id={resource.id} leading={leading(resource)}
            closeLabel={labels.close(resource.name)} onClose={onClose ? () => onClose(resource) : undefined}
            onKeyDown={(event) => { if (onClose && event.key === "Delete") { event.preventDefault(); onClose(resource); } }} />)}
        </TabsList>
      </div>
      {visible.size < resources.length && <DropdownMenu>
        <DropdownTrigger render={<Button iconOnly size="sm" variant="tertiary" aria-label={labels.more}><More aria-hidden /></Button>} />
        <DropdownContent align="end" aria-label={labels.resources} checkedIndex={resources.findIndex((resource) => resource.id === activeId)}>
          {resources.map((resource, index) => <MenuItem key={resource.id} index={index} label={resource.name} leading={leading(resource)} checked={resource.id === activeId}
            onSelect={() => { onSelect(resource.id); focusSelected(resource.id); }} />)}
        </DropdownContent>
      </DropdownMenu>}
    </div>
    {addAction && <div className="shrink-0">{addAction}</div>}
    {/* Measure the same public tabs offscreen; this copy is inert and never navigable. */}
    <div aria-hidden inert className="pointer-events-none invisible absolute inset-0 overflow-hidden">
      <div className="flex w-max items-center">
      <Tabs value={activeId} variant="segment" color="default">
        <TabsList ref={measureRef}>
          {resources.map((resource) => <TabItem key={resource.id} value={resource.id} label={resource.name} data-measure-artifact leading={leading(resource)} onClose={onClose ? () => {} : undefined} />)}
        </TabsList>
      </Tabs>
      <div ref={moreRef}><Button iconOnly size="sm" variant="tertiary" tabIndex={-1}><More aria-hidden /></Button></div>
      </div>
    </div>
  </div>;
}

    setActionBusy(true);
    setNotice("");
    try { await action(); setNotice(success); } catch { setNotice(failed); }
    finally { actionLock.current = false; setActionBusy(false); }
  };
  return <>
    <div className="flex shrink-0 flex-wrap items-center gap-2 border-b-hairline border-border px-4 py-2">
      <span className="min-w-0 flex-1 truncate font-mono text-label text-fg-muted" title={resource.path ?? resource.name}>{resource.path ?? resource.name}</span>
      <div className="flex flex-wrap items-center gap-1">
        {text !== undefined && <Button size="xs" variant="ghost" leadingIcon={Copy} disabled={actionBusy} onClick={() => void run(() => navigator.clipboard.writeText(text), labels.copied, labels.copyFailed)}>{labels.copy}</Button>}
        {downloadable && <Button size="xs" variant="ghost" onClick={() => downloadArtifact(resource)}>{labels.download}</Button>}
        {url && <Button size="xs" variant="ghost" asChild><a href={url} target="_blank" rel="noopener noreferrer">{labels.openExternal}</a></Button>}
        {onReference && <Button size="xs" variant="ghost" leadingIcon={Link} disabled={actionBusy} onClick={() => void run(() => onReference(resource), labels.referenced, labels.referenceFailed)}>{labels.reference}</Button>}
      </div>
    </div>
    {(views.length > 1 || view === "diff") && <div className="flex shrink-0 flex-wrap items-center justify-between gap-2 border-b-hairline border-border px-4 py-2">
      <Tabs value={view} onValueChange={(value) => setView(value as ArtifactView)} variant="segment" color="neutral">
        <TabsList aria-label={resource.name}>{views.map((item) => <TabItem key={item} value={item} label={labels[item]} />)}</TabsList>
      </Tabs>
      {view === "diff" && <Button variant="ghost" size="xs" aria-pressed={split} onClick={() => setSplit(!split)}>{split ? labels.split : labels.unified}</Button>}
    </div>}
    {notice && <div className="px-4 py-2 text-label text-fg-muted" role="status">{notice}</div>}
    <div className="flex min-h-0 min-w-0 flex-1 flex-col overflow-auto" data-slot="artifact-viewport" tabIndex={0} aria-label={`${labels.preview}: ${resource.name}`}>
      <ArtifactPreviewBoundary key={`${view}:${split}:${resource.status ?? "ready"}`} labels={labels}>
        <ArtifactPreview resource={resource} view={view} split={split} labels={labels} renderPreview={renderPreview}
          onRetry={onRetry ? () => void run(() => onRetry(resource), "", labels.failed) : undefined} />
      </ArtifactPreviewBoundary>
    </div>
  </>;
}

function ArtifactTabPanel({ resource, active, ...props }: {
  resource: ArtifactResource; active: boolean; labels: ArtifactLabels;
} & Pick<ArtifactPanelProps, "onReference" | "onRetry" | "renderPreview">) {
  const [visited, setVisited] = useState(active);
  useEffect(() => { if (active) setVisited(true); }, [active]);
  return <TabPanel value={resource.id} keepMounted className={cn("min-h-0 min-w-0 flex-1 flex-col", active ? "flex" : "hidden")}>
    {(active || visited) && <ArtifactSurface resource={resource} {...props} />}
  </TabPanel>;
}

export function ArtifactPanel({ resources, activeId, defaultActiveId, onActiveIdChange, onClose, addAction, onReview, onReference, onRetry, renderPreview, labels: overrides, locale, className, ...props }: ArtifactPanelProps) {
  const labels = resolveArtifactLabels(locale, overrides);
  const [selectedId, setSelectedId] = useState(defaultActiveId ?? resources[0]?.id);
  const requested = activeId === undefined ? selectedId : activeId;
  const active = resources.find((resource) => resource.id === requested) ?? resources[0];
  const [reviewing, setReviewing] = useState(false);
  const [reviewError, setReviewError] = useState(false);
  const reviewLock = useRef(false);
  const root = useRef<HTMLElement>(null);
  const resolvedId = active?.id;
  useEffect(() => {
    if (activeId === undefined) setSelectedId(resolvedId);
  }, [activeId, resolvedId]);

  const select = (id: string) => {
    if (activeId === undefined) setSelectedId(id);
    onActiveIdChange?.(id);
  };
  const close = (resource: ArtifactResource) => {
    if (!onClose) return;
    const index = resources.indexOf(resource);
    const next = resource.id === active?.id ? resources[index + 1] ?? resources[index - 1] : active;
    onClose(resource);
    if (next && next.id !== active?.id) select(next.id);
    requestAnimationFrame(() => {
      const tab = Array.from(root.current?.querySelectorAll<HTMLElement>("[data-artifact-id]") ?? []).find((node) => node.dataset.artifactId === next?.id);
      tab?.focus();
    });
  };
  const pending = getPendingReviewResources(resources);
  const review = async (items: readonly ArtifactResource[], decision: ArtifactReviewDecision) => {
    if (!onReview || reviewLock.current || !items.length) return;
    reviewLock.current = true;
    setReviewing(true);
    setReviewError(false);
    try {
      const results = await Promise.allSettled(items.map((resource) => Promise.resolve().then(() => onReview(resource, decision))));
      setReviewError(results.some((result) => result.status === "rejected"));
    } finally { reviewLock.current = false; setReviewing(false); }
  };

  return <section {...props} ref={root} aria-label={props["aria-label"] ?? labels.panel} data-slot="artifact-panel" className={cn("flex size-full min-h-0 min-w-0 flex-col overflow-hidden bg-surface-floating text-fg-default", className)}>
    {active ? <Tabs value={active.id} onValueChange={select} variant="segment" color="default" className="flex min-h-0 flex-1 flex-col">
      <ArtifactTabs resources={resources} activeId={active.id} labels={labels} onSelect={select} onClose={onClose ? close : undefined} addAction={addAction} />
      {resources.map((resource) => <ArtifactTabPanel key={artifactVersionKey(resource)} resource={resource} active={resource.id === active.id} labels={labels} onReference={onReference} onRetry={onRetry} renderPreview={renderPreview} />)}
    </Tabs> : <>
      {addAction && <div className="flex shrink-0 items-center border-b-hairline border-border bg-surface-floating p-2">{addAction}</div>}
      <div className="flex flex-1 items-center justify-center p-6"><Empty reason="first-use" scope="inline" density="compact"><EmptyHeader><EmptyTitle>{labels.empty}</EmptyTitle><EmptyDescription>{labels.emptyDescription}</EmptyDescription></EmptyHeader></Empty></div>
    </>}
    {onReview && pending.length > 0 && <div data-slot="artifact-review-bar" className="shrink-0 p-3">
      <Alert status="info" role="group" aria-busy={reviewing}>
        <AlertTitle role="status">{reviewing ? labels.reviewing : labels.remaining(pending.length)}</AlertTitle>
        <AlertAction><Button variant="ghost" disabled={reviewing} onClick={() => void review(pending, "reject")}>{labels.rejectAll}</Button><Button variant="neutral" disabled={reviewing} onClick={() => void review(pending, "accept")}>{labels.acceptAll}</Button></AlertAction>
      </Alert>
    </div>}
    {reviewError && <div className="shrink-0 p-3"><InlineNotice variant="emphasized" tone="danger" role="alert"><InlineNoticeContent>{labels.reviewFailed}</InlineNoticeContent></InlineNotice></div>}
  </section>;
}

type PanelHandle = NonNullable<ResizablePanelProps["panelRef"]> extends Ref<infer T> ? T : never;

export function ChatWorkspace({
  artifacts, title, panelOpen, defaultPanelOpen = true, onPanelOpenChange,
  activePane, defaultActivePane = "chat", onActivePaneChange,
  chatDefaultSize = "38%", chatMinSize = "280px", artifactMinSize = "320px",
  onLayoutChanged, workspaceLabels, className, ...threadProps
}: ChatWorkspaceProps) {
  const labels = resolveArtifactLabels(threadProps.locale, workspaceLabels);
  const [internalOpen, setInternalOpen] = useState(defaultPanelOpen);
  const [internalPane, setInternalPane] = useState(defaultActivePane);
  const open = panelOpen ?? internalOpen;
  const pane = activePane ?? internalPane;
  const [wide, setWide] = useState(true);
  const root = useRef<HTMLDivElement>(null);
  const chat = useRef<PanelHandle>(null);
  const desktopSize = useRef(chatDefaultSize);
  const chatId = useId();
  const artifactId = useId();
  const PanelIcon = useIcon("rectangle-horizontal");

  useEffect(() => {
    const node = root.current;
    if (!node) return;
    const observer = new ResizeObserver(() => setWide(node.clientWidth >= 760));
    setWide(node.clientWidth >= 760);
    observer.observe(node);
    return () => observer.disconnect();
  }, []);

  useEffect(() => {
    chat.current?.resize(wide ? open ? desktopSize.current : "100%" : open && pane === "artifacts" ? "0%" : "100%");
  }, [wide, open, pane]);

  const setOpen = (next: boolean) => {
    if (panelOpen === undefined) setInternalOpen(next);
    onPanelOpenChange?.(next);
  };
  const setPane = (next: "chat" | "artifacts") => {
    if (activePane === undefined) setInternalPane(next);
    onActivePaneChange?.(next);
    if (next === "artifacts") setOpen(true);
  };
  const hideChat = !wide && open && pane === "artifacts";
  const hideArtifacts = !open || (!wide && pane === "chat");
  useEffect(() => {
    if (!hideChat) return;
    const frame = requestAnimationFrame(() => root.current?.querySelector<HTMLElement>('[data-slot="artifact-panel"] [role="tab"][aria-selected="true"]')?.focus());
    return () => cancelAnimationFrame(frame);
  }, [hideChat]);

  return <div ref={root} data-block="chat-workspace-01" data-layout={wide ? "horizontal" : "single"} className={cn("flex size-full min-h-0 min-w-0 flex-col overflow-hidden bg-surface-floating text-fg-default", className)}>
    <div className="flex shrink-0 items-center justify-between gap-3 border-b-hairline border-border px-4 py-2">
      <span className="truncate text-label font-medium">{title ?? labels.workspace}</span>
      {wide ? <Button variant="ghost" size="sm" leadingIcon={PanelIcon} aria-expanded={open} aria-controls={artifactId} onClick={() => setOpen(!open)}>{open ? labels.hidePanel : labels.showPanel}</Button> :
        <div className="flex items-center gap-1">
          <Button variant="ghost" size="sm" active={!hideChat} aria-pressed={!hideChat} onClick={() => setPane("chat")}>{labels.conversation}</Button>
          <Button variant="ghost" size="sm" active={hideChat} aria-pressed={hideChat} onClick={() => setPane("artifacts")}>{labels.preview}</Button>
        </div>}
    </div>
    <ResizablePanelGroup orientation="horizontal" disabled={!wide || !open} aria-label={labels.workspace} className="min-h-0 flex-1"
      onLayoutChanged={(layout, meta) => {
        if (meta.isUserInteraction && wide && open && layout[chatId] > 10 && layout[chatId] < 90) desktopSize.current = `${layout[chatId]}%`;
        onLayoutChanged?.(layout, meta);
      }}>
      <ResizablePanel id={chatId} panelRef={chat} defaultSize={chatDefaultSize} minSize={wide && open ? chatMinSize : "0%"} className={cn("overflow-hidden", hideChat && "invisible")} inert={hideChat} aria-hidden={hideChat || undefined}>
        <ChatThread {...threadProps} />
      </ResizablePanel>
      <ResizableHandle withHandle aria-label={labels.resize} className={(!wide || !open) ? "hidden" : undefined} />
      <ResizablePanel id={artifactId} minSize={wide && open ? artifactMinSize : "0%"} className={cn("overflow-hidden", hideArtifacts && "invisible")} inert={hideArtifacts} aria-hidden={hideArtifacts || undefined}>
        {artifacts}
      </ResizablePanel>
    </ResizablePanelGroup>
  </div>;
}
