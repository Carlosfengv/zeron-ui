export function windowColor(percent: number | null) {
  if (percent === null || !Number.isFinite(percent)) return "var(--surface-raised)";
  return percent >= 100 ? activityChartColors.incident : activityColors[percent < 40 ? 1 : percent < 60 ? 2 : percent < 80 ? 3 : 4];
}

export function MatrixChart({ selection, days, calendar, labels }: {
  selection: ActivitySelection; days: ActivityDay[]; calendar: ActivityCalendar; labels: AccountActivityLabels;
}) {
  const byHour = new Map(selection.hours.map((point) => [`${point.date}/${point.hour}`, point.active]));
  const timestamps = new Map(selection.hours.map((point) => [`${point.date}/${point.hour}`, point.timestamp]));
  const columns = days.map((day, column) => ({ bin: column, bins: Array.from({ length: 24 }, (_, hour) => ({
    bin: hour, date: new Date(timestamps.get(`${day.date}/${hour}`) ?? calendar.fromCalendar(day.date, hour)), count: byHour.get(`${day.date}/${hour}`) ?? NaN,
  })) }));
  const points = selection.events.flatMap((event) => {
    const part = calendar.parts(event.timestamp), column = days.findIndex((day) => day.date === part.date);
    return column < 0 ? [] : [{ column, row: part.hour, label: `${labels.ranOut} · ${calendar.date(event.timestamp)}` }];
  });
  return <HeatmapMatrix ariaLabel={labels.matrix} data={columns} points={points} height={MATRIX_HEIGHT} minWidth={640} className="pb-2"
    columns={days.map((day, index) => ({ id: day.date, label: calendar.weekday(day.weekday).replace(/^(星期|周)/, "").slice(0, 1),
      groupLabel: index % 7 === 0 ? calendar.date(day.timestamp) : undefined, weekend: day.weekday >= 5,
      locked: selection.locks.filter((event) => event.timestamp < day.end && event.unlockedAt > day.timestamp).length }))}
    rowLabels={Array.from({ length: 24 }, (_, hour) => hour % 6 === 0 ? `${String(hour).padStart(2, "0")}:00` : "")}
    lockedLabel={labels.locked} profile={{ title: labels.dayProfile, description: labels.averageActive,
      content: <DayProfile selection={selection} height={MATRIX_HEIGHT} label={labels.dayProfile} eventHours={selection.events.map((event) => calendar.parts(event.timestamp).hour)} />,
      footer: <><span>0</span><span>{selection.accounts.length.toFixed(1)}</span></> }}
    formatHeader={(_, date) => calendar.date(date.getTime())}
    formatLabel={(count, date) => `${String(calendar.parts(date.getTime()).hour).padStart(2, "0")}:00 · ${Number.isFinite(count) ? count : labels.unknown} ${labels.active}`} />;
}

type LockInterval = { id: string; start: number; end: number };
function EventAnnotations({ events, locks, start, end }: { events: readonly AccountActivityEvent[]; locks?: readonly LockInterval[]; start: number; end: number }) {
  const { xScale, innerHeight } = useChartStable();
  const intervals = locks ?? events.map((event) => ({ id: event.id, start: event.timestamp, end: event.unlockedAt }));
  return <g pointerEvents="none">
    {intervals.filter((lock) => lock.start < end && lock.end > start).map((lock) => <rect key={lock.id}
      data-slot="account-activity-lock" x={xScale(new Date(Math.max(start, lock.start)))} y={Math.max(0, innerHeight - 3)}
      width={Math.max(0, xScale(new Date(Math.min(end, lock.end))) - xScale(new Date(Math.max(start, lock.start))))}
      height={3} fill={activityChartColors.incident} opacity={.65} />)}
    {events.filter((event) => event.timestamp >= start && event.timestamp < end).map((event) => {
      const x = xScale(new Date(event.timestamp));
      return <g key={event.id}><line x1={x} x2={x} y1={7} y2={innerHeight} stroke={activityChartColors.incident} strokeWidth={1.5} /><circle data-slot="account-activity-run-out" cx={x} cy={6} r={3.5} fill={activityChartColors.incident} /></g>;
    })}
  </g>;
}

import type { AccountActivityData, AccountActivityDayType, AccountActivityEvent, AccountActivityQuery, AccountActivitySample } from "./account-activity-types";

export const HOUR = 3_600_000;
export const DAY = HOUR * 24;
export function createActivityCalendar(timeZone: string, locale: string) {
  const parts = new Intl.DateTimeFormat("en-CA", { timeZone, year: "numeric", month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit", hourCycle: "h23" });
  const dateFormat = new Intl.DateTimeFormat(locale, { timeZone, month: "short", day: "numeric" });
  const weekdayFormat = new Intl.DateTimeFormat(locale, { timeZone: "UTC", weekday: "short" });
  return {
    fromCalendar(date: string, hour = 0) {
      const target = Date.parse(`${date}T00:00:00Z`) + hour * HOUR;
      let guess = target;
      for (let iteration = 0; iteration < 3; iteration++) {
        const fields = Object.fromEntries(parts.formatToParts(guess).map((part) => [part.type, part.value]));
        const observed = Date.parse(`${fields.year}-${fields.month}-${fields.day}T00:00:00Z`) + Number(fields.hour) * HOUR + Number(fields.minute) * 60_000;
        guess += target - observed;
      }
      return guess;
    },
    parts(timestamp: number) {
      const fields = Object.fromEntries(parts.formatToParts(timestamp).map((part) => [part.type, part.value]));
      const date = `${fields.year}-${fields.month}-${fields.day}`;
      return { date, hour: Number(fields.hour), weekday: (new Date(`${date}T00:00:00Z`).getUTCDay() + 6) % 7 };
    },
    /** Wall-clock coordinate for weekly lanes; retain partial-hour lock endpoints. */
    coordinate(timestamp: number) {
      const fields = Object.fromEntries(parts.formatToParts(timestamp).map((part) => [part.type, part.value]));
      return Date.parse(`${fields.year}-${fields.month}-${fields.day}T00:00:00Z`) + Number(fields.hour) * HOUR + Number(fields.minute) * 60_000;
    },
    date: (timestamp: number) => dateFormat.format(timestamp),
    weekday: (index: number) => weekdayFormat.format(new Date(Date.UTC(2026, 8, 14 + index))),
  };
}
export type ActivityCalendar = ReturnType<typeof createActivityCalendar>;
export function matchesDay(timestamp: number, type: AccountActivityDayType, calendar: ActivityCalendar) {
  const weekend = calendar.parts(timestamp).weekday >= 5;
  return type === "all" || (type === "weekends" ? weekend : !weekend);
}
export function isLocked(sample: Pick<AccountActivitySample, "accountId" | "timestamp">, events: readonly AccountActivityEvent[]) {
  return events.some((event) => event.accountId === sample.accountId && sample.timestamp >= event.timestamp && sample.timestamp < event.unlockedAt);
}
export function clippedLockedHours(events: readonly AccountActivityEvent[], start: number, end: number) {
  // Merge overlapping incidents per account so one blocked hour is counted once.
  const groups = new Map<string, [number, number][]>();
  for (const event of events) {
    const a = Math.max(start, event.timestamp), b = Math.min(end, event.unlockedAt);
    if (b > a) groups.set(event.accountId, [...(groups.get(event.accountId) ?? []), [a, b]]);
  }
  let total = 0;
  for (const intervals of groups.values()) {
    intervals.sort((a, b) => a[0] - b[0]);
    let edge = -Infinity;
    for (const [a, b] of intervals) { total += Math.max(0, b - Math.max(a, edge)); edge = Math.max(edge, b); }
  }
  return total / HOUR;
}
export function selectActivity(data: AccountActivityData, query: AccountActivityQuery, calendar: ActivityCalendar, selectedRange = false) {
  const start = selectedRange ? Math.max(data.start, query.range.start) : data.start;
  const end = selectedRange ? Math.min(data.end, query.range.end) : data.end;
  const hidden = new Set(query.hiddenAccountIds);
  const accounts = data.accounts.filter((account) => (query.accountId === "all" || query.accountId === account.id) && !hidden.has(account.id));
  const ids = new Set(accounts.map((account) => account.id));
  // Account readings usually share hourly timestamps. Resolve each timestamp
  // once per selection instead of formatting it again for every account.
  const calendarParts = new Map<number, ReturnType<ActivityCalendar["parts"]>>();
  const partsAt = (timestamp: number) => {
    let part = calendarParts.get(timestamp);
    if (!part) { part = calendar.parts(timestamp); calendarParts.set(timestamp, part); }
    return part;
  };
  const accepts = (timestamp: number) => {
    if (!Number.isFinite(timestamp) || timestamp < start || timestamp >= end) return false;
    if (query.dayType === "all") return true;
    const weekend = partsAt(timestamp).weekday >= 5;
    return query.dayType === "weekends" ? weekend : !weekend;
  };
  const samples = data.samples.filter((sample) => ids.has(sample.accountId) && accepts(sample.timestamp));
  const events = data.events.filter((event) => ids.has(event.accountId) && accepts(event.timestamp));
  const locks = data.events.filter((event) => ids.has(event.accountId) && event.timestamp < end && event.unlockedAt > start);
  // One observed hour contributes once even when an account reports multiple samples.
  const buckets = new Map<string, { date: string; hour: number; weekday: number; timestamp: number; values: Map<string, boolean> }>();
  for (const sample of samples) {
    if (sample.value === null || !Number.isFinite(sample.value) || sample.value < 0) continue;
    const part = partsAt(sample.timestamp), key = `${part.date}/${part.hour}`;
    let bucket = buckets.get(key);
    if (!bucket) { bucket = { ...part, timestamp: sample.timestamp, values: new Map() }; buckets.set(key, bucket); }
    bucket.values.set(sample.accountId, bucket.values.get(sample.accountId) === true || sample.value > 0);
  }
  const hours = [...buckets.values()].map(({ values, ...bucket }) => ({ ...bucket, active: [...values.values()].filter(Boolean).length }));
  const lockRanges = activityDays({ start, end }, calendar).flatMap((day) => {
    if (!matchesDay(day.timestamp, query.dayType, calendar)) return [];
    return [{ start: Math.max(start, day.timestamp), end: Math.min(end, day.end) }];
  });
  return { start, end, accounts, samples, events, locks, hours, lockRanges };
}
export type ActivitySelection = ReturnType<typeof selectActivity>;
export function average(values: readonly number[]) { return values.length ? values.reduce((sum, value) => sum + value, 0) / values.length : null; }
export function profile(selection: ActivitySelection) {
  return Array.from({ length: 24 }, (_, hour) => ({ timestamp: hour * HOUR,
    weekday: average(selection.hours.filter((point) => point.hour === hour && point.weekday < 5).map((point) => point.active)),
    weekend: average(selection.hours.filter((point) => point.hour === hour && point.weekday >= 5).map((point) => point.active)),
  }));
}
export function activityDays(data: Pick<AccountActivityData, "start" | "end">, calendar: ActivityCalendar) {
  const days: { date: string; timestamp: number; end: number; weekday: number }[] = [];
  if (!Number.isFinite(data.start) || !Number.isFinite(data.end) || data.end <= data.start) return days;
  const first = Date.parse(`${calendar.parts(data.start).date}T00:00:00Z`);
  const last = Date.parse(`${calendar.parts(data.end - 1).date}T00:00:00Z`);
  for (let date = first; date <= last; date += DAY) {
    const iso = new Date(date).toISOString().slice(0, 10), timestamp = calendar.fromCalendar(iso);
    // Some zones have historically skipped an entire calendar date.
    if (calendar.parts(timestamp).date !== iso) continue;
    days.push({ date: iso, timestamp, end: calendar.fromCalendar(new Date(date + DAY).toISOString().slice(0, 10)), weekday: (new Date(date).getUTCDay() + 6) % 7 });
  }
  return days;
}
export type ActivityDay = ReturnType<typeof activityDays>[number];

/** Brush and keyboard ranges must also fit reports shorter than one hour. */
export function clampActivityRange(data: Pick<AccountActivityData, "start" | "end">, range: AccountActivityQuery["range"]) {
  const span = Math.min(HOUR, data.end - data.start);
  const start = Math.max(data.start, Math.min(data.end - span, Number.isFinite(range.start) ? range.start : data.start));
  const end = Math.max(start + span, Math.min(data.end, Number.isFinite(range.end) ? range.end : data.end));
  return { start, end };
}

/** Keep the displayed day and three-hour peak from the same observation group. */
export function busiestActivity(selection: ActivitySelection, grouping: "dayType" | "weekday" | "all") {
  const groups = Array.from({ length: grouping === "weekday" ? 7 : grouping === "dayType" ? 2 : 1 }, () => Array.from({ length: 24 }, () => ({ sum: 0, count: 0 })));
  for (const point of selection.hours) {
    const day = grouping === "weekday" ? point.weekday : grouping === "dayType" && point.weekday >= 5 ? 1 : 0;
    const bucket = groups[day][point.hour]; bucket.sum += point.active; bucket.count++;
  }
  let best: { hour: number; day: number; score: number } | null = null;
  for (const [day, hours] of groups.entries()) {
    for (let hour = 0; hour <= 21; hour++) {
      const score = hours.slice(hour, hour + 3).reduce((sum, bucket) => sum + (bucket.count ? bucket.sum / bucket.count : 0), 0);
      if (score > 0 && (!best || score > best.score)) best = { hour, day, score };
    }
  }
  return best;
}
export function selectionLockedHours(selection: ActivitySelection, start = selection.start, end = selection.end, accountId?: string) {
  const events = accountId ? selection.locks.filter((event) => event.accountId === accountId) : selection.locks;
  return selection.lockRanges.reduce((sum, range) => sum + clippedLockedHours(events, Math.max(start, range.start), Math.min(end, range.end)), 0);
}

  const period = index === 0 || index === 5 || index === 7 ? 30 : 7;
  const origin = start - ((index * 2 + 3) % period) * DAY;
  return Array.from({ length: Math.ceil((end - origin) / (period * DAY)) }, (_, n) => {
    const a = origin + n * period * DAY, b = a + period * DAY;
    const hit = events.some((event) => event.accountId === account.id && event.timestamp >= a && event.timestamp < b);
    const current = b >= end;
    const peakPercent = hit ? 100 : Math.round((index < 6 ? 74 : 24) + ((n * 19 + index * 7) % 24));
    const observedEnd = Math.min(b, end);
    const readings = Array.from({ length: Math.ceil((observedEnd - a) / (6 * HOUR)) + 1 }, (_, slot) => {
      const timestamp = Math.min(observedEnd, a + slot * 6 * HOUR);
      return { timestamp, percent: Math.min(peakPercent, Math.round((timestamp - a) / (observedEnd - a) * peakPercent)) };
    });
    return { id: `${account.id}-${n}`, accountId: account.id, start: a, end: b, current, peakPercent, readings };
  });
});

export function AccountsView({ selection, calendar, labels, locale }: { selection: ActivitySelection; calendar: ActivityCalendar; labels: AccountActivityLabels; locale: string }) {
  const lockTotals = new Map(selection.accounts.map((account) => [account.id, selectionLockedHours(selection, selection.start, selection.end, account.id)]));
  const accounts = [...selection.accounts].sort((a, b) => (lockTotals.get(b.id) ?? 0) - (lockTotals.get(a.id) ?? 0));
  const { values, lockedHours } = useMemo(() => {
    const buckets = new Map<string, number[]>();
    const lockedHours = new Set<string>();
    for (const point of selection.samples) {
      const key = `${point.accountId}/${calendar.parts(point.timestamp).hour}`;
      if (isLocked(point, selection.locks)) { lockedHours.add(key); continue; }
      if (point.value === null || !Number.isFinite(point.value) || point.value < 0) continue;
      buckets.set(key, [...(buckets.get(key) ?? []), point.value]);
    }
    return { values: new Map([...buckets].map(([key, list]) => [key, average(list)])), lockedHours };
  }, [selection, calendar]);
  const maxByAccount = new Map(accounts.map((account) => [account.id, Math.max(0, ...Array.from({ length: 24 }, (_, hour) => values.get(`${account.id}/${hour}`) ?? 0)) || 1]));
  const rowOrigin = Date.UTC(2026, 8, 14);
  const accountAt = (date: Date) => accounts[Math.floor((date.getTime() - rowOrigin) / DAY)];
  const columns = Array.from({ length: 24 }, (_, hour) => ({ bin: hour, bins: accounts.map((account, row) => ({ bin: row, date: new Date(rowOrigin + row * DAY + hour * HOUR), count: values.get(`${account.id}/${hour}`) == null ? NaN : (values.get(`${account.id}/${hour}`) ?? 0) / (maxByAccount.get(account.id) ?? 1) * 8 })) }));
  const points = selection.events.map((event) => ({ column: calendar.parts(event.timestamp).hour, row: accounts.findIndex((account) => account.id === event.accountId) }));
  const locks = accounts.flatMap((account, row) => Array.from({ length: 24 }, (_, hour) => ({ column: hour, row })).filter((cell) => lockedHours.has(`${account.id}/${cell.column}`)));
  const top = Array.from({ length: 24 }, (_, hour) => ({ timestamp: hour * HOUR, value: average(selection.hours.filter((point) => point.hour === hour).map((point) => point.active)) }));
  const peakValue = Math.max(0, ...top.map((point) => point.value ?? 0)), max = Math.max(1, peakValue), format = new Intl.NumberFormat(locale, { maximumFractionDigits: 1 });
  return <div className="space-y-6">
    <div className="overflow-x-auto"><div className="grid min-w-192 grid-cols-[12rem_minmax(0,1fr)_8rem] items-center gap-x-5 gap-y-4">
      <div className="text-body text-fg-default">{labels.allAccounts}<p className="mt-1 text-label text-fg-subtle">{labels.averageActive}</p></div>
      <div className="space-y-2"><ActivityArea points={top} start={0} end={23 * HOUR} max={max} height={80} typical labels={labels} /><HourAxis /></div>
      <span className="text-body text-fg-subtle">{labels.peak} {format.format(peakValue)}<p className="text-label">{labels.active}</p></span>
      <div className="flex flex-col" style={{ height: accounts.length * 34 }}>{accounts.map((account) => <div className="flex h-full min-h-0 items-center" key={account.id}><AccountName account={account} /></div>)}</div>
      <ActivityHeatmap columns={columns} points={points} locks={locks} height={accounts.length * 34} labels={labels}
        formatHeader={(_, date) => accountAt(date)?.name}
        formatLabel={(_, date) => { const account = accountAt(date), value = account ? values.get(`${account.id}/${date.getUTCHours()}`) : null; return `${String(date.getUTCHours()).padStart(2, "0")}:00 · ${value == null ? labels.unknown : format.format(value)}`; }} />
      <div className="flex flex-col" style={{ height: accounts.length * 34 }}>{accounts.map((account) => {
        const hours = lockTotals.get(account.id) ?? 0;
        return <span key={account.id} className={cn("flex h-full items-center justify-end whitespace-nowrap text-body", hours ? "text-fg-default" : "text-fg-subtle")}>{hours ? labels.lockedHours(format.format(hours)) : labels.neverLocked}</span>;
      })}</div>
    </div></div><ActivityLegend labels={labels} relative /><p className="text-label text-fg-subtle">{labels.scaleHint}</p>
  </div>;
}

  const incidents = selection?.events ?? [];
  const mostLimited = selection?.accounts.map((account) => ({ account, events: incidents.filter((event) => event.accountId === account.id) })).sort((a, b) => b.events.length - a.events.length)[0];
  const eventHours = mostLimited?.events.map((event) => calendar.parts(event.timestamp).hour) ?? [];
  const eventTime = eventHours.length ? `${String(Math.min(...eventHours)).padStart(2, "0")}:00–${String(Math.min(24, Math.max(...eventHours) + 1)).padStart(2, "0")}:00` : "";
  const title = state === "loading" ? labels.loading : state === "error" || !validBounds ? labels.error : query.view === "range" ? `${calendar.date(query.range.start)} – ${calendar.date(query.range.end - 1)}`
    : query.view === "accounts" && mostLimited?.events.length ? labels.accountSummary(mostLimited.account.name, eventTime)
    : busiest ? labels.busiest(busiest.time, query.view === "weeks" ? calendar.weekday(busiest.day) : busiest.day === 1 ? labels.weekends : labels.weekdays) : selection?.hours.length ? labels.quietActivity : labels.empty;
  const description = state !== "ready" || !validBounds ? "" : query.view === "range" ? `${labels.rangeSummary(selection?.accounts.length ?? 0)} · ${labels.rangeHint}`
    : query.view === "accounts" && busiest ? labels.busiest(busiest.time, labels.allAccounts)
    : labels.summary(incidents.length, days.length, incidents.filter((event) => calendar.parts(event.timestamp).weekday < 5).length);

  return <section {...props} aria-labelledby={titleId} data-slot="account-activity" data-view={query.view} data-theme-transition="instant" className={cn("min-w-0 w-full text-body", className)}
    style={{ "--zeron-theme-color-duration": "0ms", ...style } as CSSProperties}>
    <Card className="border-hairline border-border">
      <CardContent className="p-4 sm:p-6">
        <Tabs value={query.view} onValueChange={(view) => { if (views.includes(view as AccountActivityView)) update({ view: view as AccountActivityView }); }} variant="segment" color="default" className="min-w-0 space-y-7">
          <div className="flex flex-wrap items-start justify-between gap-4">
            <div className="min-w-0 flex-1 basis-80"><h2 id={titleId} className="text-title font-medium text-fg-default sm:text-heading">{title}</h2><p className="mt-1 text-label text-fg-subtle" aria-live="polite">{description}</p></div>
            <div className="flex max-w-full flex-wrap items-center gap-3">
              {query.view === "accounts" ? <Select value={query.dayType} onValueChange={(dayType) => update({ dayType: dayType as AccountActivityQuery["dayType"] })}>
                <SelectTrigger aria-label={labels.dayFilter} className="w-36" /><SelectContent>{(["weekdays", "weekends", "all"] as const).map((day) => <SelectItem key={day} value={day}>{labels[day]}</SelectItem>)}</SelectContent>
              </Select> : <Select value={query.accountId} onValueChange={(accountId) => update({ accountId })}>
                <SelectTrigger aria-label={labels.accountFilter} className="w-44" /><SelectContent><SelectItem value="all">{labels.allAccounts}</SelectItem>{data.accounts.map((account) => <SelectItem key={account.id} value={account.id}>{account.name}</SelectItem>)}</SelectContent>
              </Select>}
              <div className="max-w-full overflow-x-auto"><TabsList aria-label={labels.views}>{views.map((view) => <TabItem key={view} value={view} label={labels[view]} />)}</TabsList></div>
            </div>
          </div>
          {state === "loading" ? <div role="status" className="flex min-h-80 items-center justify-center text-body text-fg-subtle">{labels.loading}</div>
            : state === "error" || !validBounds ? <div role="alert" className="flex min-h-80 flex-col items-center justify-center gap-4 text-body text-fg-muted"><p>{errorMessage ?? labels.error}</p>{onRetry && <Button variant="secondary" onClick={onRetry}>{labels.retry}</Button>}</div>
            : selection && <>
              <TabPanel value="matrix"><div className="space-y-6">{selection.hours.length ? <MatrixChart selection={selection} days={days} calendar={calendar} labels={labels} /> : <EmptyState text={labels.empty} />}<ActivityLegend labels={labels} /><WindowUsage data={data} selection={selection} calendar={calendar} labels={labels} /></div></TabPanel>
              <TabPanel value="weeks">{selection.hours.length ? <WeeksView data={data} selection={selection} calendar={calendar} labels={labels} locale={locale} /> : <EmptyState text={labels.empty} />}</TabPanel>
              <TabPanel value="range"><RangeView data={data} selection={selection} calendar={calendar} labels={labels} query={query} onChange={update} /></TabPanel>
              <TabPanel value="accounts">{selection.hours.length ? <AccountsView selection={selection} calendar={calendar} labels={labels} locale={locale} /> : <EmptyState text={labels.empty} />}</TabPanel>
            </>}
        </Tabs>
      </CardContent>
    </Card>
  </section>;
}

function EmptyState({ text }: { text: string }) { return <p role="status" className="flex min-h-80 items-center justify-center text-body text-fg-subtle">{text}</p>; }
